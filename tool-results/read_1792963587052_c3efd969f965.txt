    57→/** تشخیص تایپ‌کردن در فیلدهای متنی — کلیدهای جهت‌نما نباید صحنه را حرکت دهند */
    58→function isTypingTarget(t: EventTarget | null): boolean {
    59→  const el = t as HTMLElement | null;
    60→  if (!el) return false;
    61→  return el.isContentEditable || ['INPUT', 'TEXTAREA', 'SELECT'].includes(el.tagName);
    62→}

    64→/** کارت ورود مجدد وقتی نشست مدیر منقضی شده (401) */
    65→function OwnerLoginInline({ onDone }: { onDone: () => void }) {
    66→  const { toast } = useToast();
    67→  const [pw, setPw] = useState('');
    68→  const [busy, setBusy] = useState(false);
    69→
    70→  const submit = async () => {
    71→    if (!pw || busy) return;
    72→    setBusy(true);
    73→    try {
    74→      await api.login(pw);
    75→      onDone();
    76→    } catch (e) {
    77→      toast({
    78→        title: 'ورود ناموفق',
    79→        description: (e as Error).message || 'رمز مدیر اشتباه است',
    80→        variant: 'destructive',
    81→      });
    82→    } finally {
    83→      setBusy(false);
    84→    }
    85→  };
    86→
    87→  return (
    88→    <div className="h-screen flex items-center justify-center bg-stone-50 p-6">
    89→      <Card className="w-full max-w-sm shadow-xl">
    90→        <CardContent className="p-8 space-y-5 text-center">
    91→          <div className="mx-auto h-14 w-14 rounded-2xl bg-primary/10 flex items-center justify-center">
    92→            <Lock className="h-7 w-7 text-primary" />
    93→          </div>
    94→          <div className="space-y-1.5">
    95→            <h1 className="text-lg font-bold">ورود مدیر</h1>
    96→            <p className="text-sm text-muted-foreground">نشست شما منقضی شده — برای ادامه رمز را وارد کنید</p>
    97→          </div>
    98→          <Input
    99→            type="password"
   100→            value={pw}
   101→            onChange={(e) => setPw(e.target.value)}
   102→            onKeyDown={(e) => e.key === 'Enter' && submit()}
   103→            placeholder="رمز مدیر"
   104→            autoFocus
   105→            className="h-11 text-center"
   106→            aria-label="رمز مدیر"
   107→          />
   108→          <Button className="w-full h-11" onClick={submit} disabled={!pw || busy}>
   109→            {busy ? 'در حال ورود...' : 'ورود'}
   110→          </Button>
   111→        </CardContent>
   112→      </Card>
   113→    </div>
   114→  );
   115→}

   503→          aria-label="نام پروژه"
   504→        />
   505→
   506→        <div className="flex items-center gap-1">
   507→          <Button

   685→
   686→          {dropActive && (
   687→            <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
   688→              <Badge className="text-sm px-4 py-2 shadow-lg">رها کنید تا اضافه شود</Badge>
