   324→   324→  let lastError: unknown = null;
   325→   325→  for (let attempt = 1; attempt <= 2; attempt++) {

   328→   328→        visionBody(RECONSTRUCT_PROMPT, image),
   329→   329→      );

   331→   331→      if (typeof content !== 'string' || !content.trim()) {
   332→   332→        throw new Error('پاسخ خالی از مدل');
   333→   333→      }
   334→   334→      const analysis = normalizeAnalysis(extractJsonObject(content));
   335→   335→      if (!analysis) throw new Error('خروجی مدل آبجکت JSON معتبر نبود');
   336→   336→      return analysis;
   337→   337→    } catch (err) {
   338→   338→      lastError = err;
   339→   339→      console.error(`[reconstruct] vision attempt ${attempt} (image ${index}) failed:`, err);
   340→   340→    }
   341→   341→  }
   342→   342→  console.error(`[reconstruct] all vision attempts failed for image ${index}:`, lastError);
   343→   343→  return null;
   344→   344→}

   372→   372→      : JSON.stringify(partials);
   373→   373→    const completion = await zai.chat.completions.create({
   374→   374→      messages: [
   375→   375→        { role: 'system', content: MERGE_PROMPT },
   376→   376→        { role: 'user', content: userContent },
   377→   377→      ] satisfies ChatMessage[],
   378→   378→      thinking: { type: 'disabled' },
   379→   379→    });
   380→   380→    const content = completion?.choices?.[0]?.message?.content;
   381→   381→    if (typeof content === 'string' && content.trim()) {
   382→   382→      const merged = normalizeAnalysis(extractJsonObject(content));
   383→   383→      if (merged) return merged;
   384→   384→    }
   385→   385→    throw new Error('خروجی ادغام نامعتبر بود');
   386→   386→  } catch (err) {
   387→   387→    console.error('[reconstruct] merge failed, using first partial:', err);
   388→   388→    return partials[0];
   389→   389→  }
   390→   390→}

   418→   418→/** روشنایی نسبی رنگ hex (۰ تا ۱) — برای جلوگیری از شیشه تیره */
   419→   419→function luminance(hex: string): number {
   420→   420→  const r = parseInt(hex.slice(1, 3), 16);
   421→   421→  const g = parseInt(hex.slice(3, 5), 16);
   422→   422→  const b = parseInt(hex.slice(5, 7), 16);
   423→   423→  return (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   424→   424→}

   426→   426→/** رنگ تیره را برای قاب شیشه به‌قدر کافی روشن می‌کند */
   427→   427→const lightenIfDark = (hex: string): string =>
   428→   428→  luminance(hex) < 0.35 ? shadeColor(hex, 0.65) : hex;

   430→   430→/** آفست را طوری محدود می‌کند که آیتم داخل دیوار بماند */
   431→   431→function clampOffsetOnWall(wall: WallName, offsetX: number, itemWidth: number, room: Analysis['room']) {
   432→   432→  const wallLength = wall === 'back' || wall === 'front' ? room.widthM : room.depthM;
   433→   433→  const maxAbs = wallLength / 2 - 0.05 - itemWidth / 2;
   434→   434→  return maxAbs <= 0 ? 0 : clamp(offsetX, -maxAbs, maxAbs);
   435→   435→}

   464→   464→      handles: 'bar', mountY: 1.5,
   465→   465→    };

   471→   471→  } else if (kind === 'bed') {

   473→   473→  } else if (kind === 'table') {
   474→   474→    rawParams = { w, h: 0.75, d: clamp(d, 0.7, 1.2), color, bodyColor: shadeColor(color, -0.1) };
   475→   475→  } else if (kind === 'chair') {
   476→   476→    rawParams = { w: 0.45, h: 0.9, d: 0.5, color, bodyColor: color };
   477→   477→  } else if (kind === 'washer') {
   478→   478→    rawParams = { w: 0.6, h: 0.85, d: 0.6, color, bodyColor: color };
   479→   479→  } else if (kind === 'dishwasher') {
   480→   480→    rawParams = { w: 0.6, h: 0.85, d: 0.6, color, bodyColor: color, doors: 1, handles: 'bar' };
   481→   481→  } else if (kind === 'microwave') {
   482→   482→    // مایکروویو روی کانتر (y = 0.9)
   483→   483→    rawParams = { w: clamp(w, 0.4, 0.7), h: 0.3, d: 0.38, color, bodyColor: color, mountY: 0.9 };
   484→   484→  } else {
   485→   485→    // base / tall / island / shelf
   486→   486→    rawParams = {
   487→   487→      w, h, d, bodyColor: color, doorColor: shadeColor(color, 0.2),
   488→   488→      countertopColor: '#d8d3c8', doors: w > 0.7 ? 2 : 1, drawers: 0, handles: 'bar',
   489→   489→    };
   490→   490→  }
   491→   491→
   492→   492→  // درب شیشه‌ای — فقط انواع کابینتی؛ قاب شیشه هرگز تیره نمی‌ماند
   493→   493→  if (GLASS_KINDS.has(kind)) {
   494→   494→    rawParams.glass = item.glassDoors === true;
   495→   495→    if (item.glassDoors) {
   496→   496→      if (typeof rawParams.bodyColor === 'string') rawParams.bodyColor = lightenIfDark(rawParams.bodyColor);
   497→   497→      if (typeof rawParams.doorColor === 'string') rawParams.doorColor = lightenIfDark(rawParams.doorColor);
   498→   498→    }
   499→   499→  }
   500→   500→
   501→   501→  const params = normalizePartialParams(rawParams, kind);
   502→   502→  const y = kind === 'wall' || kind === 'hood' || kind === 'tv' || kind === 'microwave'
   503→   503→    ? params.mountY ?? 0
   504→   504→    : 0;

   506→   506→  const { position, rotationY } = placeAtWall(
   507→   507→    item.wall,
   508→   508→    offsetX,
   509→   509→    d,
   510→   510→    { widthM: room.widthM, depthM: room.depthM },

   512→   512→  );

   534→   534→/**
   535→   535→ * خط لوله کامل: تحلیل موازی همه عکس‌ها + نقشه → ادغام → ساخت آبجکت‌ها.
   536→   536→ * send در حالت استریم رویدادهای progress را می‌نویسد؛ در حالت JSON undefined است.
   537→   537→ */
   538→   538→async function runPipeline(
   539→   539→  zai: ZAI,
   540→   540→  images: string[],
   541→   541→  floorPlanImage: string | undefined,
   542→   542→  send?: SendFn,
   543→   543→): Promise<ReconstructResult> {
   544→   544→  // نقشه در همان دسته موازی اجرا می‌شود؛ خطایش نادیده گرفته می‌شود
   545→   545→  const floorPlanPromise: Promise<FloorPlanSummary | null> | null = floorPlanImage
   546→   546→    ? (async () => {
   547→   547→        send?.({ type: 'progress', stage: 'floorplan', message: 'در حال بررسی نقشه ساختمان...' });
   548→   548→        const fp = await analyzeFloorPlan(zai, floorPlanImage);
   549→   549→        if (fp) {
   550→   550→          send?.({ type: 'progress', stage: 'floorplan', message: 'نقشه ساختمان بررسی شد' });
   551→   551→        }
   552→   552→        return fp;
   553→   553→      })()
   554→   554→    : null;
   555→   555→
   556→   556→  // تحلیل موازی تصاویر — بعد از اتمام هر عکس یک رویداد progress می‌آید
   557→   557→  const total = images.length;
   558→   558→  const imagePromises = images.map((img, i) =>
   559→   559→    analyzeImage(zai, img, i).then((analysis) => {
   560→   560→      send?.({
   561→   561→        type: 'progress',
   562→   562→        stage: 'analyze',
   563→   563→        index: i + 1,
   564→   564→        total,
   565→   565→        message: `تحلیل عکس ${faDigits(i + 1)} از ${faDigits(total)} انجام شد`,
   566→   566→      });
   567→   567→      return analysis;
   568→   568→    }),
   569→   569→  );
   570→   570→
   571→   571→  // همه در یک Promise.allSettled — نقشه اول صف است (اگر بوده باشد)
   572→   572→  const tasks: Promise<unknown>[] = [
   573→   573→    ...(floorPlanPromise ? [floorPlanPromise] : []),
   574→   574→    ...imagePromises,
   575→   575→  ];
   576→   576→  const settled = await Promise.allSettled(tasks);
   577→   577→  const offset = floorPlanPromise ? 1 : 0;
   578→   578→
   579→   579→  const floorPlanSummary = floorPlanPromise
   580→   580→    ? settled[0].status === 'fulfilled'
   581→   581→      ? (settled[0].value as FloorPlanSummary | null)
   582→   582→      : null
   583→   583→    : null;
   584→   584→
   585→   585→  const partials = settled
   586→   586→    .slice(offset)
   587→   587→    .map((r) => (r.status === 'fulfilled' ? (r.value as Analysis | null) : null))
   588→   588→    .filter((a): a is Analysis => !!a);
   589→   589→
   590→   590→  if (partials.length === 0) {
   591→   591→    throw new ReconstructFailure('تحلیل تصاویر با هوش مصنوعی ناموفق بود');
   592→   592→  }
   593→   593→
   594→   594→  // مرحله ادغام (با نقشه به‌عنوان ground truth)
   595→   595→  if (partials.length > 1) {
   596→   596→    send?.({ type: 'progress', stage: 'merge', message: 'ادغام نتایج تحلیل‌ها...' });
   597→   597→  }
   598→   598→  const merged = await mergeAnalyses(zai, partials, floorPlanSummary);
   599→   599→
   600→   600→  const roomSpec: RoomSpec = {
   601→   601→    ...merged.room,
   602→   602→    windows: merged.windows,
   603→   603→    doorways: merged.doorways,
   604→   604→  };
   605→   605→  const objects = merged.items.map((item) => buildObject(item, merged.room));
   606→   606→
   607→   607→  return {
   608→   608→    roomType: merged.roomType,
   609→   609→    roomSpec,
   610→   610→    objects,
   611→   611→    description: merged.description,
   612→   612→  };
   613→   613→}

   633→   633→// ---------------- استریم NDJSON ----------------

   635→   635→function ndjsonResponse(run: (send: SendFn) => Promise<void>): NextResponse {
   636→   636→  const encoder = new TextEncoder();
   637→   637→  const stream = new ReadableStream<Uint8Array>({
   638→   638→    async start(controller) {
   639→   639→      let closed = false;
   640→   640→      const send: SendFn = (obj) => {
   641→   641→        if (closed) return;
   642→   642→        try {
   643→   643→          controller.enqueue(encoder.encode(JSON.stringify(obj) + '\n'));
   644→   644→        } catch {
   645→   645→          // کلاینت قطع شده — بی‌خطر
   646→   646→        }
   647→   647→      };
   648→   648→      try {
   649→   649→        await run(send);
   650→   650→      } catch (err) {
   651→   651→        const message =
   652→   652→          err instanceof ReconstructFailure
   653→   653→            ? err.message
   654→   654→            : 'بازسازی با هوش مصنوعی ناموفق بود';
   655→   655→        if (!(err instanceof ReconstructFailure)) {
   656→   656→          console.error('[reconstruct] stream failed:', err);
   657→   657→        }
   658→   658→        send({ type: 'error', error: message });
   659→   659→      } finally {
   660→   660→        closed = true;
   661→   661→        try {
   662→   662→          controller.close();
   663→   663→        } catch {
   664→   664→          // قبلاً بسته شده
   665→   665→        }
   666→   666→      }
   667→   667→    },
   668→   668→  });
   669→   669→  return new NextResponse(stream, {
   670→   670→    headers: {
   671→   671→      'content-type': 'application/x-ndjson',
   672→   672→      'cache-control': 'no-cache, no-transform',
   673→   673→    },
   674→   674→  });
   675→   675→}

   679→   679→export async function POST(req: NextRequest) {
   680→   680→  // گارد مالک — قبل از هر چیز (AI گران است)
   681→   681→  const denied = await requireOwner(req);
   682→   682→  if (denied) return denied;
   683→   683→
   684→   684→  // حالت JSON (سازگار با نسخه ۲) در صورت Accept: application/json؛ پیش‌فرض استریم NDJSON
   685→   685→  const wantsJson = (req.headers.get('accept') ?? '').includes('application/json');
   686→   686→
   687→   687→  const body = await readJsonBody(req);
   688→   688→  if (!body.ok) {
   689→   689→    return NextResponse.json({ error: 'داده ارسالی نامعتبر است' }, { status: 400 });
   690→   690→  }
   691→   691→
   692→   692→  try {
   693→   693→    const parsed = bodySchema.safeParse(body.data);
   694→   694→    if (!parsed.success) {
   695→   695→      return NextResponse.json(
   696→   696→        { error: `حداکثر ${MAX_IMAGES} تصویر و حداقل یک تصویر لازم است` },
   697→   697→        { status: 400 },
   698→   698→      );
   699→   699→    }
   700→   700→
   701→   701→    // سازگاری با نسخه ۱: { image } → { images: [image] }
   702→   702→    const images = (parsed.data.images ?? (parsed.data.image ? [parsed.data.image] : []))
   703→   703→      .map((s) => s.trim())
   704→   704→      .filter(Boolean);
   705→   705→    if (images.length === 0) {
   706→   706→      return NextResponse.json({ error: 'تصویری ارسال نشده است' }, { status: 400 });
   707→   707→    }
   708→   708→
   709→   709→    // اعتبارسنجی همه تصاویر — هر تصویر نامعتبر کل درخواست را رد می‌کند
   710→   710→    for (const image of images) {
   711→   711→      const invalid = validateImage(image);
   712→   712→      if (invalid) return invalid;
   713→   713→    }
   714→   714→    const floorPlan = parsed.data.floorPlan?.trim() || undefined;
   715→   715→    if (floorPlan) {
   716→   716→      const invalid = validateImage(floorPlan);
   717→   717→      if (invalid) return invalid;
   718→   718→    }
   719→   719→
   720→   720→    const zai = await ZAI.create();
   721→   721→
   722→   722→    if (wantsJson) {
   723→   723→      try {
   724→   724→        const result = await runPipeline(zai, images, floorPlan);
   725→   725→        return NextResponse.json(result);
   726→   726→      } catch (err) {
   727→   727→        if (err instanceof ReconstructFailure) {
   728→   728→          return NextResponse.json({ error: err.message }, { status: 502 });
   729→   729→        }
   730→   730→        throw err;
   731→   731→      }
   732→   732→    }
   733→   733→
   734→   734→    // حالت استریم — رویدادهای progress و در پایان result یا error
   735→   735→    return ndjsonResponse(async (send) => {
   736→   736→      const result = await runPipeline(zai, images, floorPlan, send);
   737→   737→      send({ type: 'result', ...result });
   738→   738→    });
   739→   739→  } catch (err) {
   740→   740→    console.error('POST /api/reconstruct failed:', err);
   741→   741→    return NextResponse.json(
   742→   742→      { error: 'بازسازی با هوش مصنوعی ناموفق بود' },
   743→   743→      { status: 502 },
   744→   744→    );
   745→   745→  }
   746→   746→}
