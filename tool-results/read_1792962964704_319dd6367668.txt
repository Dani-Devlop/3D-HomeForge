   461→            <p className="text-xs text-primary bg-primary/5 border border-primary/15 rounded-md px-3 py-2 leading-relaxed" aria-live="polite">

   467→              <div

   478→    </div>

   501→  const [name, setName] = useState('');
   502→  const [recSteps, setRecSteps] = useState<string[]>([]);
   503→  const [progress, setProgress] = useState<{ step: number; percent: number; message: string } | null>(null);
   504→  const stepsRef = useRef<string[]>([]);
   505→
   506→  const buildSteps = (hasPlan: boolean) => {
   507→    const steps = ['ارسال تصاویر به موتور هوش مصنوعی...'];
   508→    if (hasPlan) steps.push('خواندن پلان خانه برای ابعاد دقیق...');
   509→    steps.push('تحلیل عکس‌ها و تشخیص فضا و تجهیزات...');
   510→    steps.push('ادغام نتایج چند زاویه...');
   511→    steps.push('ساخت مش سه‌بعدی ویرایش‌پذیر...');
   512→    return steps;
   513→  };
   514→
   515→  const addImages = async (files: FileList | null) => {
   516→    if (!files?.length) return;
   517→    const imgs = Array.from(files).filter((f) => f.type.startsWith('image/'));
   518→    if (!imgs.length) {
   519→      toast({ title: 'فقط تصویر مجاز است', variant: 'destructive' });
   520→      return;
   521→    }
   522→    try {
   523→      const remaining = MAX_SHOTS - shots.length;
   524→      if (remaining <= 0) {
   525→        toast({ title: `حداکثر ${MAX_SHOTS.toLocaleString('fa-IR')} عکس مجاز است`, variant: 'destructive' });
   526→        return;
   527→      }
   528→      const urls = await Promise.all(imgs.slice(0, remaining).map((f) => fileToResizedDataUrl(f)));
   529→      setShots((prev) => {
   530→        const merged = [
   531→          ...prev.filter((s) => s.key !== videoKey || !videoKey),
   532→          ...urls.map((src, i) => ({ key: `img-${Date.now()}-${i}`, src })),
   533→        ];
   534→        return merged.slice(0, MAX_SHOTS);
   535→      });
   536→    } catch {
   537→      toast({ title: 'خواندن تصویر ناموفق بود', variant: 'destructive' });
   538→    }
   539→  };
   540→
   541→  const addVideo = async (f: File | null | undefined) => {
   542→    if (!f) return;
   543→    if (!f.type.startsWith('video/')) {
   544→      toast({ title: 'فایل ویدیویی معتبر نیست', variant: 'destructive' });
   545→      return;
   546→    }
   547→    try {
   548→      toast({ title: 'در حال استخراج فریم‌های ویدیو...' });
   549→      const frames = await videoToFrames(f, 3);
   550→      const vk = `vid-${Date.now()}`;
   551→      setVideoKey(vk);
   552→      setShots((prev) =>
   553→        [...prev.filter((s) => !s.key.startsWith('vid-')), ...frames.map((src, i) => ({ key: `${vk}-${i}`, src }))].slice(0, MAX_SHOTS)
   554→      );
   555→    } catch (e) {
   556→      toast({
   557→        title: 'استخراج فریم ناموفق بود',
   558→        description: e instanceof Error ? e.message : undefined,
   559→        variant: 'destructive',
   560→      });
   561→    }
   562→  };
   563→
   564→  const addFloorPlan = async (f: File | null | undefined) => {
   565→    if (!f) return;
   566→    if (!f.type.startsWith('image/')) {
   567→      toast({ title: 'پلان باید تصویر باشد', variant: 'destructive' });
   568→      return;
   569→    }
   570→    try {
   571→      setFloorPlan(await fileToResizedDataUrl(f, 1600));
   572→    } catch {
   573→      toast({ title: 'خواندن پلان ناموفق بود', variant: 'destructive' });
   574→    }
   575→  };
   576→
   577→  const reconstruct = useMutation({
   578→    mutationFn: async () => {
   579→      if (!shots.length) throw new Error('حداقل یک عکس یا ویدیو انتخاب کنید');
   580→      const hasPlan = !!floorPlan;
   581→      const steps = buildSteps(hasPlan);
   582→      stepsRef.current = steps;
   583→      setRecSteps(steps);
   584→      setProgress({ step: 0, percent: 5, message: '' });
   585→      const idx = {
   586→        floorplan: hasPlan ? 1 : 0,
   587→        analyze: hasPlan ? 2 : 1,
   588→        merge: hasPlan ? 3 : 2,
   589→      };
   590→      return api.reconstructStream(
   591→        shots.map((s) => s.src),
   592→        floorPlan,
   593→        (p) => {
   594→          if (p.stage === 'floorplan') {
   595→            setProgress({ step: idx.floorplan, percent: 10, message: p.message });
   596→          } else if (p.stage === 'analyze') {
   597→            const frac = p.total ? Math.min(1, (p.index ?? 0) / p.total) : 0;
   598→            setProgress({ step: idx.analyze, percent: 15 + frac * 65, message: p.message });
   599→          } else if (p.stage === 'merge') {
   600→            setProgress({ step: idx.merge, percent: 85, message: p.message });
   601→          } else {
   602→            setProgress((prev) => ({
   603→              step: prev?.step ?? 0,
   604→              percent: prev?.percent ?? 5,
   605→              message: p.message,
   606→            }));
   607→          }
   608→        },
   609→      );
   610→    },
   611→    onSuccess: async (res) => {
   612→      // گام آخر فقط بعد از موفقیت کامل می‌شود — هرگز قبل از آن گیر نمی‌کند
   613→      setProgress({ step: stepsRef.current.length - 1, percent: 100, message: '' });
   614→      const created = await api.createProject({
   615→        name: name.trim() || 'پروژه بازسازی‌شده',
   616→        roomType: res.roomType,
   617→        roomSpec: res.roomSpec,
   618→        sceneState: { objects: res.objects },
   619→        folderId,
   620→      });
   621→      qc.invalidateQueries({ queryKey: ['projects'] });
   622→      toast({
   623→        title: 'مدل سه‌بعدی ساخته شد ✓',
   624→        description: res.objects.length
   625→          ? `${res.objects.length.toLocaleString('fa-IR')} قطعه تجهیزات/کابینت موجود شناسایی شد`
   626→          : res.description || undefined,
   627→      });
   628→      window.location.hash = `/editor/${created.project.id}`;
   629→    },
   630→    onError: (e: Error) => {
   631→      setProgress(null);
   632→      setRecSteps([]);
   633→      toast({
   634→        title: 'بازسازی ناموفق بود',
   635→        description: `${e.message} — می‌توانید با فضای نمونه شروع کنید`,
   636→        variant: 'destructive',
   637→      });
   638→    },
   639→  });
   640→
   641→  const startSample = useMutation({
   642→    mutationFn: () =>
   643→      api.createProject({
   644→        name: name.trim() || 'آشپزخانه نمونه',
   645→        roomType: 'kitchen',
   646→        roomSpec: SAMPLE_ROOM_SPEC,
   647→        sceneState: SAMPLE_SCENE,
   648→        folderId,
   649→      }),
   650→    onSuccess: (res) => {
   651→      qc.invalidateQueries({ queryKey: ['projects'] });
   652→      window.location.hash = `/editor/${res.project.id}`;
   653→    },
   654→    onError: (e: Error) =>
   655→      toast({ title: 'خطا', description: e.message, variant: 'destructive' }),
   656→  });
   657→
   658→  return (
   659→    <Card className="w-full">
   660→      <CardContent className="p-5 space-y-4">
   661→        <div className="flex items-center gap-2">
   662→          <Camera className="h-5 w-5 text-primary" />
   663→          <h2 className="font-bold">بازسازی دقیق فضا از عکس یا ویدیو</h2>
   664→        </div>
   665→
   666→        <input
   667→          ref={imgRef}
   668→          type="file"
   669→          accept="image/*"
   670→          multiple
   671→          className="hidden"
   672→          onChange={(e) => {
   673→            addImages(e.target.files);
   674→            e.target.value = '';
   675→          }}
   676→          aria-hidden
   677→        />
   678→        <input
   679→          ref={vidRef}
   680→          type="file"
   681→          accept="video/*"
   682→          className="hidden"
   683→          onChange={(e) => {
   684→            addVideo(e.target.files?.[0]);
   685→            e.target.value = '';
   686→          }}
   687→          aria-hidden
   688→        />
   689→        <input
   690→          ref={planRef}
   691→          type="file"
   692→          accept="image/*"
   693→          className="hidden"
   694→          onChange={(e) => {
   695→            addFloorPlan(e.target.files?.[0]);
   696→            e.target.value = '';
   697→          }}
   698→          aria-hidden
   699→        />
   700→
   701→        <div
   702→          role="button"
   703→          tabIndex={0}
   704→          onClick={() => imgRef.current?.click()}
   705→          onKeyDown={(e) => {
   706→            if (e.key === 'Enter' || e.key === ' ') {
   707→              e.preventDefault();
   708→              imgRef.current?.click();
   709→            }
   710→          }}
   711→          onDragOver={(e) => e.preventDefault()}
   712→          onDrop={(e) => {
   713→            e.preventDefault();
   714→            addImages(e.dataTransfer.files);
   715→          }}
   716→          className="w-full rounded-xl border-2 border-dashed border-border hover:border-primary/60 bg-muted/30 transition-colors p-6 flex flex-col items-center gap-2 cursor-pointer group min-h-[120px] justify-center outline-none focus-visible:ring-2 focus-visible:ring-primary/50"
   717→          aria-label="انتخاب عکس‌های فضا"
   718→        >
   719→          {shots.length === 0 ? (
   720→            <>
   721→              <UploadCloud className="h-9 w-9 text-muted-foreground/50 group-hover:text-primary transition-colors" />
   722→              <p className="text-sm font-medium">چند عکس از زوایای مختلف بکشید یا انتخاب کنید</p>
   723→              <p className="text-xs text-muted-foreground">
   724→                روبه‌رو، چپ و راست — پوشش کامل ۳۶۰ درجه لازم نیست
   725→              </p>
   726→            </>
   727→          ) : (
   728→            <div className="grid grid-cols-4 gap-2 w-full">
   729→              {shots.map((s) => (
   730→                <div key={s.key} className="relative group/th rounded-lg overflow-hidden border">
   731→                  <img src={s.src} alt="نمای آپلودشده فضا" className="w-full h-16 object-cover" />
   732→                  <button
   733→                    type="button"
   734→                    onClick={(e) => {
   735→                      e.stopPropagation();
   736→                      setShots((prev) => prev.filter((x) => x.key !== s.key));
   737→                    }}
   738→                    className="absolute top-1 end-1 h-5 w-5 rounded-full bg-black/60 text-white flex items-center justify-center opacity-0 group-hover/th:opacity-100 transition-opacity"
   739→                    aria-label="حذف تصویر"
   740→                  >
   741→                    <X className="h-3 w-3" />
   742→                  </button>
   743→                </div>
   744→              ))}
   745→              {shots.length < MAX_SHOTS && (
   746→                <div className="h-16 rounded-lg border border-dashed flex flex-col items-center justify-center text-[10px] text-muted-foreground gap-0.5">
   747→                  <ImageIcon className="h-4 w-4" />
   748→                  افزودن (تا {(MAX_SHOTS - shots.length).toLocaleString('fa-IR')})
   749→                </div>
   750→              )}
   751→            </div>
   752→          )}
   753→        </div>
   754→
   755→        <div className="grid grid-cols-2 gap-2">
   756→          <Button
   757→            type="button"
   758→            variant="outline"
   759→            size="sm"
   760→            className="h-9 gap-1.5 text-xs"
   761→            onClick={() => imgRef.current?.click()}
   762→            disabled={shots.length >= MAX_SHOTS}
   763→          >
   764→            <ImageIcon className="h-3.5 w-3.5" /> عکس‌های بیشتر
   765→          </Button>
   766→          <Button
   767→            type="button"
   768→            variant="outline"
   769→            size="sm"
   770→            className="h-9 gap-1.5 text-xs"
   771→            onClick={() => vidRef.current?.click()}
   772→            disabled={shots.length >= MAX_SHOTS - 1}
   773→          >
   774→            <Video className="h-3.5 w-3.5" /> آپلود ویدیو
   775→          </Button>
   776→        </div>
   777→
   778→        {/* پلان خانه — اختیاری */}
   779→        {floorPlan ? (
   780→          <div className="flex items-center gap-3 rounded-xl border bg-muted/30 p-2">
   781→            <img src={floorPlan} alt="پلان خانه" className="h-14 w-20 object-cover rounded-lg border shrink-0" />
   782→            <div className="min-w-0 flex-1">
   783→              <p className="text-xs font-medium flex items-center gap-1.5">
   784→                <Ruler className="h-3.5 w-3.5 text-primary" /> پلان خانه افزوده شد
   785→              </p>
   786→              <p className="text-[10px] text-muted-foreground">ابعاد از روی پلان خوانده می‌شود</p>
   787→            </div>
   788→            <Button
   789→              size="icon"
   790→              variant="ghost"
   791→              className="h-8 w-8 shrink-0"
   792→              onClick={() => setFloorPlan(null)}
   793→              aria-label="حذف پلان"
   794→            >
   795→              <X className="h-4 w-4" />
   796→            </Button>
   797→          </div>
   798→        ) : (
   799→          <div
   800→            role="button"
   801→            tabIndex={0}
   802→            onClick={() => planRef.current?.click()}
   803→            onKeyDown={(e) => {
   804→              if (e.key === 'Enter' || e.key === ' ') {
   805→                e.preventDefault();
   806→                planRef.current?.click();
   807→              }
   808→            }}
   809→            onDragOver={(e) => e.preventDefault()}
   810→            onDrop={(e) => {
   811→              e.preventDefault();
   812→              addFloorPlan(e.dataTransfer.files?.[0]);
   813→            }}
   814→            className="w-full rounded-xl border-2 border-dashed border-primary/25 hover:border-primary/60 bg-primary/5 transition-colors p-3 flex items-center gap-3 cursor-pointer outline-none focus-visible:ring-2 focus-visible:ring-primary/50"
   815→            aria-label="افزودن پلان خانه (اختیاری)"
   816→          >
   817→            <Ruler className="h-6 w-6 text-primary/60 shrink-0" />
   818→            <div className="min-w-0">
   819→              <p className="text-xs font-medium">پلان خانه (اختیاری — برای دقت بیشتر)</p>
   820→              <p className="text-[10px] text-muted-foreground leading-relaxed">
   821→                لازم نیست همیشه پلان بدهید — فقط وقتی عکس پلان دارید اضافه کنید تا ابعاد دقیق‌تر شود
   822→              </p>
   823→            </div>
   824→          </div>
   825→        )}
   826→
   827→        <Input
   828→          value={name}
   829→          onChange={(e) => setName(e.target.value)}
   830→          placeholder="نام پروژه (اختیاری)"
   831→          maxLength={80}
   832→        />
   833→
   834→        <div className="flex flex-col sm:flex-row gap-2">
   835→          <Button
   836→            className="flex-1 gap-2 h-11"
   837→            onClick={() => reconstruct.mutate()}
   838→            disabled={!shots.length || reconstruct.isPending}
   839→          >
   840→            <Sparkles className="h-4 w-4" />
   841→            {reconstruct.isPending ? 'در حال بازسازی...' : 'بازسازی سه‌بعدی با AI'}
   842→          </Button>
   843→          <Button
   844→            variant="outline"
   845→            className="sm:w-auto gap-2 h-11"
   846→            onClick={() => startSample.mutate()}
   847→            disabled={startSample.isPending}
   848→          >
   849→            <Boxes className="h-4 w-4" />
   850→            فضای نمونه
   851→          </Button>
   852→        </div>
   853→        <p className="text-[11px] text-muted-foreground leading-relaxed">
   854→          AI دیوارها، بازشوها و تجهیزات موجود (کابینت، یخچال، اجاق، هود، مبلمان) را تشخیص می‌دهد و مدل
   855→          سه‌بعدی تقریبی اما قابل ویرایش می‌سازد.
   856→        </p>
   857→      </CardContent>
   858→
   859→      {progress && (
   860→        <ReconstructOverlay
   861→          steps={recSteps}
   862→          step={progress.step}
   863→          percent={progress.percent}
   864→          message={progress.message}
   865→        />
   866→      )}
   867→    </Card>
   868→  );
   869→}
