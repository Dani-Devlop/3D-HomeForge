   214→/** خروجی مدل را به تحلیل امن تبدیل می‌کند؛ ورودی غیرآبجکت → null */
   215→function normalizeAnalysis(data: unknown): Analysis | null {
   216→  if (!isRecord(data)) return null;
   217→  const room = isRecord(data.room) ? data.room : {};
   218→
   219→  const windows = (Array.isArray(data.windows) ? data.windows : [])
   220→    .slice(0, 8)
   221→    .filter(isRecord)
   222→    .map((w) => ({
   223→      wall: toWall(w.wall),
   224→      offsetX: clamp(toFinite(w.offsetX, 0), -7, 7),
   225→      sillY: clamp(toFinite(w.sillY, 1), 0.3, 1.8),
   226→      width: clamp(toFinite(w.width, 1.2), 0.6, 2.6),
   227→      height: clamp(toFinite(w.height, 1.2), 0.6, 2.2),
   228→    }));
   229→
   230→  const doorways = (Array.isArray(data.doorways) ? data.doorways : [])
   231→    .slice(0, 4)
   232→    .filter(isRecord)
   233→    .map((d) => ({
   234→      wall: toWall(d.wall),
   235→      offsetX: clamp(toFinite(d.offsetX, 0), -7, 7),
   236→      width: clamp(toFinite(d.width, 0.95), 0.8, 1.4),
   237→      height: clamp(toFinite(d.height, 2.1), 1.9, 2.5),
   238→    }));
   239→
   240→  const items = (Array.isArray(data.items) ? data.items : [])
   241→    .slice(0, MAX_OBJECTS)
   242→    .filter(isRecord)
   243→    .map((it): ReconstructItem => {
   244→      const dRaw = toFinite(it.depthM, Number.NaN);
   245→      return {
   246→        kind: toFurnitureKind(it.kind),
   247→        wall: toWall(it.wall),
   248→        offsetX: clamp(toFinite(it.offsetX, 0), -7, 7),
   249→        widthM: clamp(toFinite(it.widthM, 0.8), 0.3, 3),
   250→        heightM: clamp(toFinite(it.heightM, 0.85), 0.3, 2.4),
   251→        depthM: Number.isFinite(dRaw) ? clamp(dRaw, 0.25, 1.2) : undefined,
   252→        glassDoors: it.glassDoors === true,
   253→        colorHex: toColor(it.colorHex, '#6b4f3a'),
   254→        description: typeof it.description === 'string' ? it.description.slice(0, 120) : '',
   255→      };
   256→    });
   257→
   258→  return {
   259→    roomType:
   260→      typeof data.roomType === 'string' && ROOM_TYPES.includes(data.roomType)
   261→        ? data.roomType
   262→        : 'other',
   263→    room: {
   264→      widthM: clamp(toFinite(room.widthM, 4), 2, 15),
   265→      depthM: clamp(toFinite(room.depthM, 3.5), 2, 15),
   266→      heightM: clamp(toFinite(room.heightM, 2.8), 2.2, 4.2),
   267→      wallColor: toColor(room.wallColor, '#ece7dd'),
   268→      floorColor: toColor(room.floorColor, '#a97c50'),
   269→      ceilingColor: toColor(room.ceilingColor, '#fafafa'),
   270→    },
   271→    windows,
   272→    doorways,
   273→    items,
   274→    description: typeof data.description === 'string' ? data.description.slice(0, 600) : '',
   275→  };
   276→}

   278→/** نرمال‌سازی تدافعی خروجی نقشه — خطای نقشه هرگز کل درخواست را نمی‌شکند */
   279→function normalizeFloorPlan(data: unknown): FloorPlanSummary | null {
   280→  if (!isRecord(data)) return null;
   281→  return {
   282→    summary: typeof data.summary === 'string' ? data.summary.slice(0, 1200) : '',
   283→    widthM: optFinite(data.widthM, 2, 30),
   284→    depthM: optFinite(data.depthM, 2, 30),
   285→    rooms: (Array.isArray(data.rooms) ? data.rooms : [])
   286→      .slice(0, 20)
   287→      .filter(isRecord)
   288→      .map((r) => ({
   289→        name: typeof r.name === 'string' ? r.name.slice(0, 60) : '',
   290→        widthM: optFinite(r.widthM, 0.5, 30),
   291→        depthM: optFinite(r.depthM, 0.5, 30),
   292→      })),
   293→    windows: (Array.isArray(data.windows) ? data.windows : [])
   294→      .slice(0, 12)
   295→      .filter(isRecord),
   296→    notes: (Array.isArray(data.notes) ? data.notes : [])
   297→      .slice(0, 12)
   298→      .filter((n): n is string => typeof n === 'string')
   299→      .map((n) => n.slice(0, 200)),
   300→  };
   301→}

   322→/** تحلیل VLM یک تصویر با حداکثر ۲ تلاش */
   323→async function analyzeImage(zai: ZAI, image: string, index: number): Promise<Analysis | null> {
   324→  let lastError: unknown = null;
   325→  for (let attempt = 1; attempt <= 2; attempt++) {
   326→    try {
   327→      const visionResult = await zai.chat.completions.createVision(
   328→        visionBody(RECONSTRUCT_PROMPT, image),
   329→      );
   330→      const content = visionResult?.choices?.[0]?.message?.content;
   331→      if (typeof content !== 'string' || !content.trim()) {
   332→        throw new Error('پاسخ خالی از مدل');
   333→      }
   334→      const analysis = normalizeAnalysis(extractJsonObject(content));
   335→      if (!analysis) throw new Error('خروجی مدل آبجکت JSON معتبر نبود');
   336→      return analysis;
   337→    } catch (err) {
   338→      lastError = err;
   339→      console.error(`[reconstruct] vision attempt ${attempt} (image ${index}) failed:`, err);
   340→    }
   341→  }
   342→  console.error(`[reconstruct] all vision attempts failed for image ${index}:`, lastError);
   343→  return null;
   344→}

   361→/** ادغام تحلیل‌های جزئی (+ نقشه) با یک فراخوانی متنی؛ در صورت خطا اولین تحلیل برگردانده می‌شود */
   362→async function mergeAnalyses(
   363→  zai: ZAI,
   364→  partials: Analysis[],
   365→  floorPlan: FloorPlanSummary | null,
   366→): Promise<Analysis> {
   367→  if (partials.length === 1 && !floorPlan) return partials[0];
   368→  try {
   369→    // نقشه به‌عنوان ground truth همراه پارتیال‌ها داده می‌شود
   370→    const userContent = floorPlan
   371→      ? JSON.stringify({ floorPlan, parts: partials })
   372→      : JSON.stringify(partials);
   373→    const completion = await zai.chat.completions.create({
   374→      messages: [
   375→        { role: 'system', content: MERGE_PROMPT },
   376→        { role: 'user', content: userContent },
   377→      ] satisfies ChatMessage[],
   378→      thinking: { type: 'disabled' },
   379→    });
   380→    const content = completion?.choices?.[0]?.message?.content;
   381→    if (typeof content === 'string' && content.trim()) {
   382→      const merged = normalizeAnalysis(extractJsonObject(content));
   383→      if (merged) return merged;
   384→    }
   385→    throw new Error('خروجی ادغام نامعتبر بود');
   386→  } catch (err) {
   387→    console.error('[reconstruct] merge failed, using first partial:', err);
   388→    return partials[0];
   389→  }
   390→}

   418→/** روشنایی نسبی رنگ hex (۰ تا ۱) — برای جلوگیری از شیشه تیره */
   419→function luminance(hex: string): number {
   420→  const r = parseInt(hex.slice(1, 3), 16);
   421→  const g = parseInt(hex.slice(3, 5), 16);
   422→  const b = parseInt(hex.slice(5, 7), 16);
   423→  return (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   424→}

   426→/** رنگ تیره را برای قاب شیشه به‌قدر کافی روشن می‌کند */
   427→const lightenIfDark = (hex: string): string =>
   428→  luminance(hex) < 0.35 ? shadeColor(hex, 0.65) : hex;
   429→
   430→/** آفست را طوری محدود می‌کند که آیتم داخل دیوار بماند */
   431→function clampOffsetOnWall(wall: WallName, offsetX: number, itemWidth: number, room: Analysis['room']) {
   432→  const wallLength = wall === 'back' || wall === 'front' ? room.widthM : room.depthM;
   433→  const maxAbs = wallLength / 2 - 0.05 - itemWidth / 2;
   434→  return maxAbs <= 0 ? 0 : clamp(offsetX, -maxAbs, maxAbs);
   435→}

   437→function buildObject(item: ReconstructItem, room: Analysis['room']): SceneObject {
   438→  const kind = item.kind;
   439→  const w = item.widthM;
   440→  const h = item.heightM;
   441→  const d = item.depthM ?? KIND_DEFAULT_D[kind];
   442→  const color = item.colorHex;
   443→
   444→  let rawParams: Record<string, unknown>;
   445→  if (kind === 'fridge') {
   446→    rawParams = {
   447→      w, h, d, color, bodyColor: color, doorColor: shadeColor(color, 0.2),
   448→      doors: 1, drawers: 0, handles: 'bar',
   449→    };
   450→  } else if (kind === 'stove') {
   451→    rawParams = {
   452→      w, h, d, color, bodyColor: color, doorColor: shadeColor(color, 0.15),
   453→      countertopColor: '#1e2024', doors: 1, drawers: 0, handles: 'bar',
   454→    };
   455→  } else if (kind === 'hood') {
   456→    rawParams = {
   457→      w, h, d, color, bodyColor: color,
   458→      doors: 0, drawers: 0, handles: 'none', mountY: 1.62,
   459→    };
   460→  } else if (kind === 'wall') {
   461→    rawParams = {
   462→      w, h, d, bodyColor: color, doorColor: shadeColor(color, 0.2),
   463→      countertopColor: '#d8d3c8', doors: w > 0.7 ? 2 : 1, drawers: 0,
   464→      handles: 'bar', mountY: 1.5,
   465→    };
   466→  } else if (kind === 'tv') {
   467→    // تلویزیون نازک — نصب روی دیوار با ارتفاع پیش‌فرض ۱.۴ متر
   468→    rawParams = { w, h, d: 0.06, color: '#101318', bodyColor: '#1b1f26', mountY: 1.4 };
   469→  } else if (kind === 'sofa') {
   470→    rawParams = { w, h: clamp(h, 0.7, 1.0), d: clamp(d, 0.8, 1.2), color, bodyColor: shadeColor(color, -0.15) };
   471→  } else if (kind === 'bed') {
   472→    rawParams = { w, h: 0.55, d: clamp(d, 1.8, 2.2), color, bodyColor: shadeColor(color, -0.2) };
   473→  } else if (kind === 'table') {
   474→    rawParams = { w, h: 0.75, d: clamp(d, 0.7, 1.2), color, bodyColor: shadeColor(color, -0.1) };
   475→  } else if (kind === 'chair') {
   476→    rawParams = { w: 0.45, h: 0.9, d: 0.5, color, bodyColor: color };
   477→  } else if (kind === 'washer') {
   478→    rawParams = { w: 0.6, h: 0.85, d: 0.6, color, bodyColor: color };
   479→  } else if (kind === 'dishwasher') {
   480→    rawParams = { w: 0.6, h: 0.85, d: 0.6, color, bodyColor: color, doors: 1, handles: 'bar' };
   481→  } else if (kind === 'microwave') {
   482→    // مایکروویو روی کانتر (y = 0.9)
   483→    rawParams = { w: clamp(w, 0.4, 0.7), h: 0.3, d: 0.38, color, bodyColor: color, mountY: 0.9 };
   484→  } else {
   485→    // base / tall / island / shelf
   486→    rawParams = {
   487→      w, h, d, bodyColor: color, doorColor: shadeColor(color, 0.2),
   488→      countertopColor: '#d8d3c8', doors: w > 0.7 ? 2 : 1, drawers: 0, handles: 'bar',
   489→    };
   490→  }
   491→
   492→  // درب شیشه‌ای — فقط انواع کابینتی؛ قاب شیشه هرگز تیره نمی‌ماند
   493→  if (GLASS_KINDS.has(kind)) {
   494→    rawParams.glass = item.glassDoors === true;
   495→    if (item.glassDoors) {
   496→      if (typeof rawParams.bodyColor === 'string') rawParams.bodyColor = lightenIfDark(rawParams.bodyColor);
   497→      if (typeof rawParams.doorColor === 'string') rawParams.doorColor = lightenIfDark(rawParams.doorColor);
   498→    }
   499→  }
   500→
   501→  const params = normalizePartialParams(rawParams, kind);
   502→  const y = kind === 'wall' || kind === 'hood' || kind === 'tv' || kind === 'microwave'
   503→    ? params.mountY ?? 0
   504→    : 0;
   505→  const offsetX = clampOffsetOnWall(item.wall, item.offsetX, w, room);
   506→  const { position, rotationY } = placeAtWall(
   507→    item.wall,
   508→    offsetX,
   509→    d,
   510→    { widthM: room.widthM, depthM: room.depthM },
   511→    y,
   512→  );
   513→
   514→  return {
   515→    id: crypto.randomUUID().slice(0, 8),
   516→    type: kind as ObjectType,
   517→    label: RECONSTRUCT_LABELS[kind],
   518→    params,
   519→    position,
   520→    rotationY,
   521→    ...(kind === 'tv' ? { on: false } : {}),
   522→  };
   523→}

   525→// ---------------- خط لوله مشترک (JSON و استریم) ----------------
   526→
   527→interface ReconstructResult {
   528→  roomType: string;
   529→  roomSpec: RoomSpec;
   530→  objects: SceneObject[];
   531→  description: string;
   532→}
   533→
   534→/**
   535→ * خط لوله کامل: تحلیل موازی همه عکس‌ها + نقشه → ادغام → ساخت آبجکت‌ها.
   536→ * send در حالت استریم رویدادهای progress را می‌نویسد؛ در حالت JSON undefined است.
   537→ */
   538→async function runPipeline(
   539→  zai: ZAI,
   540→  images: string[],
   541→  floorPlanImage: string | undefined,
   542→  send?: SendFn,
   543→): Promise<ReconstructResult> {
   544→  // نقشه در همان دسته موازی اجرا می‌شود؛ خطایش نادیده گرفته می‌شود
   545→  const floorPlanPromise: Promise<FloorPlanSummary | null> | null = floorPlanImage
   546→    ? (async () => {
   547→        send?.({ type: 'progress', stage: 'floorplan', message: 'در حال بررسی نقشه ساختمان...' });
   548→        const fp = await analyzeFloorPlan(zai, floorPlanImage);
   549→        if (fp) {
   550→          send?.({ type: 'progress', stage: 'floorplan', message: 'نقشه ساختمان بررسی شد' });
   551→        }
   552→        return fp;
   553→      })()
   554→    : null;
   555→
   556→  // تحلیل موازی تصاویر — بعد از اتمام هر عکس یک رویداد progress می‌آید
   557→  const total = images.length;
   558→  const imagePromises = images.map((img, i) =>
   559→    analyzeImage(zai, img, i).then((analysis) => {
   560→      send?.({
   561→        type: 'progress',
   562→        stage: 'analyze',
   563→        index: i + 1,
   564→        total,
   565→        message: `تحلیل عکس ${faDigits(i + 1)} از ${faDigits(total)} انجام شد`,
   566→      });
   567→      return analysis;
   568→    }),
   569→  );
   570→
   571→  // همه در یک Promise.allSettled — نقشه اول صف است (اگر بوده باشد)
   572→  const tasks: Promise<unknown>[] = [
   573→    ...(floorPlanPromise ? [floorPlanPromise] : []),
   574→    ...imagePromises,
   575→  ];
   576→  const settled = await Promise.allSettled(tasks);
   577→  const offset = floorPlanPromise ? 1 : 0;
   578→
   579→  const floorPlanSummary = floorPlanPromise
   580→    ? settled[0].status === 'fulfilled'
   581→      ? (settled[0].value as FloorPlanSummary | null)
   582→      : null
   583→    : null;
   584→
   585→  const partials = settled
   586→    .slice(offset)
   587→    .map((r) => (r.status === 'fulfilled' ? (r.value as Analysis | null) : null))
   588→    .filter((a): a is Analysis => !!a);
   589→
   590→  if (partials.length === 0) {
   591→    throw new ReconstructFailure('تحلیل تصاویر با هوش مصنوعی ناموفق بود');
   592→  }
   593→
   594→  // مرحله ادغام (با نقشه به‌عنوان ground truth)
   595→  if (partials.length > 1) {
   596→    send?.({ type: 'progress', stage: 'merge', message: 'ادغام نتایج تحلیل‌ها...' });
   597→  }
   598→  const merged = await mergeAnalyses(zai, partials, floorPlanSummary);
   599→
   600→  const roomSpec: RoomSpec = {
   601→    ...merged.room,
   602→    windows: merged.windows,
   603→    doorways: merged.doorways,
   604→  };
   605→  const objects = merged.items.map((item) => buildObject(item, merged.room));
   606→
   607→  return {
   608→    roomType: merged.roomType,
   609→    roomSpec,
   610→    objects,
   611→    description: merged.description,
   612→  };
   613→}

   617→function validateImage(image: string): NextResponse | null {
   618→  if (image.length > MAX_IMAGE_LENGTH) {
   619→    return NextResponse.json(
   620→      { error: 'حجم یکی از تصاویر بیش از حد مجاز است (حداکثر ۷ مگابایت)' },
   621→      { status: 400 },
   622→    );
   623→  }
   624→  if (!DATA_URL_RE.test(image) && !RAW_BASE64_RE.test(image)) {
   625→    return NextResponse.json(
   626→      { error: 'فرمت یکی از ارسالی‌ها تصویر نیست' },
   627→      { status: 400 },
   628→    );
   629→  }
   630→  return null;
   631→}

   633→// ---------------- استریم NDJSON ----------------
   634→
   635→function ndjsonResponse(run: (send: SendFn) => Promise<void>): NextResponse {
   636→  const encoder = new TextEncoder();
   637→  const stream = new ReadableStream<Uint8Array>({
   638→    async start(controller) {
   639→      let closed = false;
   640→      const send: SendFn = (obj) => {
   641→        if (closed) return;
   642→        try {
   643→          controller.enqueue(encoder.encode(JSON.stringify(obj) + '\n'));
   644→        } catch {
   645→          // کلاینت قطع شده — بی‌خطر
   646→        }
   647→      };
   648→      try {
   649→        await run(send);
   650→      } catch (err) {
   651→        const message =
   652→          err instanceof ReconstructFailure
   653→            ? err.message
   654→            : 'بازسازی با هوش مصنوعی ناموفق بود';
   655→        if (!(err instanceof ReconstructFailure)) {
   656→          console.error('[reconstruct] stream failed:', err);
   657→        }
   658→        send({ type: 'error', error: message });
   659→      } finally {
   660→        closed = true;
   661→        try {
   662→          controller.close();
   663→        } catch {
   664→          // قبلاً بسته شده
   665→        }
   666→      }
   667→    },
   668→  });
   669→  return new NextResponse(stream, {
   670→    headers: {
   671→      'content-type': 'application/x-ndjson',
   672→      'cache-control': 'no-cache, no-transform',
   673→    },
   674→  });
   675→}

   679→export async function POST(req: NextRequest) {
   680→  // گارد مالک — قبل از هر چیز (AI گران است)
   681→  const denied = await requireOwner(req);
   682→  if (denied) return denied;
   683→
   684→  // حالت JSON (سازگار با نسخه ۲) در صورت Accept: application/json؛ پیش‌فرض استریم NDJSON
   685→  const wantsJson = (req.headers.get('accept') ?? '').includes('application/json');
   686→
   687→  const body = await readJsonBody(req);
   688→  if (!body.ok) {
   689→    return NextResponse.json({ error: 'داده ارسالی نامعتبر است' }, { status: 400 });
   690→  }
   691→
   692→  try {
   693→    const parsed = bodySchema.safeParse(body.data);
   694→    if (!parsed.success) {
   695→      return NextResponse.json(
   696→        { error: `حداکثر ${MAX_IMAGES} تصویر و حداقل یک تصویر لازم است` },
   697→        { status: 400 },
   698→      );
   699→    }
   700→
   701→    // سازگاری با نسخه ۱: { image } → { images: [image] }
   702→    const images = (parsed.data.images ?? (parsed.data.image ? [parsed.data.image] : []))
   703→      .map((s) => s.trim())
   704→      .filter(Boolean);
   705→    if (images.length === 0) {
   706→      return NextResponse.json({ error: 'تصویری ارسال نشده است' }, { status: 400 });
   707→    }
   708→
   709→    // اعتبارسنجی همه تصاویر — هر تصویر نامعتبر کل درخواست را رد می‌کند
   710→    for (const image of images) {
   711→      const invalid = validateImage(image);
   712→      if (invalid) return invalid;
   713→    }
   714→    const floorPlan = parsed.data.floorPlan?.trim() || undefined;
   715→    if (floorPlan) {
   716→      const invalid = validateImage(floorPlan);
   717→      if (invalid) return invalid;
   718→    }
   719→
   720→    const zai = await ZAI.create();
   721→
   722→    if (wantsJson) {
   723→      try {
   724→        const result = await runPipeline(zai, images, floorPlan);
   725→        return NextResponse.json(result);
   726→      } catch (err) {
   727→        if (err instanceof ReconstructFailure) {
   728→          return NextResponse.json({ error: err.message }, { status: 502 });
   729→        }
   730→        throw err;
   731→      }
   732→    }
   733→
   734→    // حالت استریم — رویدادهای progress و در پایان result یا error
   735→    return ndjsonResponse(async (send) => {
   736→      const result = await runPipeline(zai, images, floorPlan, send);
   737→      send({ type: 'result', ...result });
   738→    });
   739→  } catch (err) {
   740→    console.error('POST /api/reconstruct failed:', err);
   741→    return NextResponse.json(
   742→      { error: 'بازسازی با هوش مصنوعی ناموفق بود' },
   743→      { status: 502 },
   744→    );
   745→  }
   746→}
