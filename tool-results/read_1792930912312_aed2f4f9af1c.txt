    47→/** تشخیص تایپ‌کردن در فیلدهای متنی — کلیدهای جهت‌نما نباید صحنه را حرکت دهند */
    48→function isTypingTarget(t: EventTarget | null): boolean {
    49→  const el = t as HTMLElement | null;
    50→  if (!el) return false;
    51→  return el.isContentEditable || ['INPUT', 'TEXTAREA', 'SELECT'].includes(el.tagName);
    52→}

    54→/** کارت ورود مجدد وقتی نشست مدیر منقضی شده (401) */
    55→function OwnerLoginInline({ onDone }: { onDone: () => void }) {
    56→  const { toast } = useToast();
    57→  const [pw, setPw] = useState('');
    58→  const [busy, setBusy] = useState(false);
    59→
    60→  const submit = async () => {
    61→    if (!pw || busy) return;
    62→    setBusy(true);
    63→    try {
    64→      await api.login(pw);
    65→      onDone();
    66→    } catch (e) {
    67→      toast({
    68→        title: 'ورود ناموفق',
    69→        description: (e as Error).message || 'رمز مدیر اشتباه است',
    70→        variant: 'destructive',
    71→      });
    72→    } finally {
    73→      setBusy(false);
    74→    }
    75→  };
    76→
    77→  return (
    78→    <div className="h-screen flex items-center justify-center bg-stone-50 p-6">
    79→      <Card className="w-full max-w-sm shadow-xl">
    80→        <CardContent className="p-8 space-y-5 text-center">
    81→          <div className="mx-auto h-14 w-14 rounded-2xl bg-primary/10 flex items-center justify-center">
    82→            <Lock className="h-7 w-7 text-primary" />
    83→          </div>
    84→          <div className="space-y-1.5">
    85→            <h1 className="text-lg font-bold">ورود مدیر</h1>
    86→            <p className="text-sm text-muted-foreground">نشست شما منقضی شده — برای ادامه رمز را وارد کنید</p>
    87→          </div>
    88→          <Input
    89→            type="password"
    90→            value={pw}
    91→            onChange={(e) => setPw(e.target.value)}
    92→            onKeyDown={(e) => e.key === 'Enter' && submit()}
    93→            placeholder="رمز مدیر"
    94→            autoFocus
    95→            className="h-11 text-center"
    96→            aria-label="رمز مدیر"
    97→          />
    98→          <Button className="w-full h-11" onClick={submit} disabled={!pw || busy}>
    99→            {busy ? 'در حال ورود...' : 'ورود'}
   100→          </Button>
   101→        </CardContent>
   102→      </Card>
   103→    </div>
   104→  );
   105→}

   649→            کلیک: انتخاب • کلیدهای جهت‌نما: حرکت • Q/E: چرخش • Shift+جهت‌نما: قدم بزرگ

   651→
   652→          {/* پد حرکت آبجکت انتخاب‌شده (موبایل + دسکتاپ) */}
   653→          {primary && (

   656→              onRotate={holdRotate}
   657→              onDelete={() => store.getState().removeObject(primary.id)}
   658→              onDuplicate={() => store.getState().duplicateObject(primary.id)}
   659→            />
   660→          )}
   661→
   662→          {dropActive && (
   663→            <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
   664→              <Badge className="text-sm px-4 py-2 shadow-lg">رها کنید تا اضافه شود</Badge>
   665→            </div>
   666→          )}
   667→        </main>
   668→
   669→        {/* پنل مشخصات */}
   670→        <aside className="w-full md:w-72 shrink-0 bg-card border-t md:border-t-0 md:border-s md:order-3 h-64 md:h-auto">
