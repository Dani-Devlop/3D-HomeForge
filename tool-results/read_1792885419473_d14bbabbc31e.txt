   123→   123→function UploadCard({ folderId }: { folderId: string | null }) {

   125→   125→  const qc = useQueryClient();
   126→   126→  const imgRef = useRef<HTMLInputElement>(null);
   127→   127→  const vidRef = useRef<HTMLInputElement>(null);
   128→   128→  const [shots, setShots] = useState<Shot[]>([]);
   129→   129→  const [videoKey, setVideoKey] = useState<string | null>(null);
   130→   130→  const [name, setName] = useState('');
   131→   131→  const [aiStep, setAiStep] = useState(-1);
   132→   132→
   133→   133→  useEffect(() => {
   134→   134→    if (aiStep < 0) return;
   135→   135→    const timer = setInterval(() => {
   136→   136→      setAiStep((s) => Math.min(s + 1, AI_STEPS.length - 1));
   137→   137→    }, 1800);
   138→   138→    return () => clearInterval(timer);
   139→   139→  }, [aiStep]);
   140→   140→
   141→   141→  const addImages = async (files: FileList | null) => {
   142→   142→    if (!files?.length) return;
   143→   143→    const imgs = Array.from(files).filter((f) => f.type.startsWith('image/'));
   144→   144→    if (!imgs.length) {
   145→   145→      toast({ title: 'فقط تصویر مجاز است', variant: 'destructive' });
   146→   146→      return;
   147→   147→    }
   148→   148→    try {
   149→   149→      const urls = await Promise.all(imgs.slice(0, 4).map((f) => fileToResizedDataUrl(f)));
   150→   150→      setShots((prev) => {
   151→   151→        const merged = [
   152→   152→          ...prev.filter((s) => s.key !== videoKey || !videoKey),
   153→   153→          ...urls.map((src, i) => ({ key: `img-${Date.now()}-${i}`, src })),
   154→   154→        ];
   155→   155→        return merged.slice(0, 5);
   156→   156→      });
   157→   157→    } catch {
   158→   158→      toast({ title: 'خواندن تصویر ناموفق بود', variant: 'destructive' });
   159→   159→    }
   160→   160→  };
   161→   161→
   162→   162→  const addVideo = async (f: File | null | undefined) => {
   163→   163→    if (!f) return;
   164→   164→    if (!f.type.startsWith('video/')) {
   165→   165→      toast({ title: 'فایل ویدیویی معتبر نیست', variant: 'destructive' });
   166→   166→      return;
   167→   167→    }
   168→   168→    try {
   169→   169→      toast({ title: 'در حال استخراج فریم‌های ویدیو...' });
   170→   170→      const frames = await videoToFrames(f, 3);
   171→   171→      const vk = `vid-${Date.now()}`;
   172→   172→      setVideoKey(vk);
   173→   173→      setShots((prev) =>
   174→   174→        [...prev.filter((s) => !s.key.startsWith('vid-')), ...frames.map((src, i) => ({ key: `${vk}-${i}`, src }))].slice(0, 5)
   175→   175→      );
   176→   176→    } catch (e) {
   177→   177→      toast({
   178→   178→        title: 'استخراج فریم ناموفق بود',
   179→   179→        description: e instanceof Error ? e.message : undefined,
   180→   180→        variant: 'destructive',
   181→   181→      });
   182→   182→    }
   183→   183→  };
   184→   184→
   185→   185→  const reconstruct = useMutation({
   186→   186→    mutationFn: async () => {
   187→   187→      if (!shots.length) throw new Error('حداقل یک عکس یا ویدیو انتخاب کنید');
   188→   188→      setAiStep(0);
   189→   189→      const res = await api.reconstruct(shots.map((s) => s.src));
   190→   190→      await new Promise((r) => setTimeout(r, 800));
   191→   191→      return res;
   192→   192→    },
   193→   193→    onSuccess: async (res) => {
   194→   194→      setAiStep(AI_STEPS.length - 1);
   195→   195→      const created = await api.createProject({
   196→   196→        name: name.trim() || 'پروژه بازسازی‌شده',
   197→   197→        roomType: res.roomType,
   198→   198→        roomSpec: res.roomSpec,
   199→   199→        sceneState: { objects: res.objects },
   200→   200→        folderId,
   201→   201→      });
   202→   202→      qc.invalidateQueries({ queryKey: ['projects'] });
   203→   203→      toast({
   204→   204→        title: 'مدل سه‌بعدی ساخته شد ✓',
   205→   205→        description: res.objects.length
   206→   206→          ? `${res.objects.length} قطعه تجهیزات/کابینت موجود شناسایی شد`
   207→   207→          : res.description || undefined,
   208→   208→      });
   209→   209→      window.location.hash = `/editor/${created.project.id}`;
   210→   210→    },
   211→   211→    onError: (e: Error) => {
   212→   212→      setAiStep(-1);
   213→   213→      toast({
   214→   214→        title: 'بازسازی ناموفق بود',
   215→   215→        description: `${e.message} — می‌توانید با فضای نمونه شروع کنید`,
   216→   216→        variant: 'destructive',
   217→   217→      });
   218→   218→    },
   219→   219→  });
   220→   220→
   221→   221→  const startSample = useMutation({
   222→   222→    mutationFn: () =>
   223→   223→      api.createProject({
   224→   224→        name: name.trim() || 'آشپزخانه نمونه',
   225→   225→        roomType: 'kitchen',
   226→   226→        roomSpec: SAMPLE_ROOM_SPEC,
   227→   227→        sceneState: SAMPLE_SCENE,
   228→   228→        folderId,
   229→   229→      }),
   230→   230→    onSuccess: (res) => {
   231→   231→      qc.invalidateQueries({ queryKey: ['projects'] });
   232→   232→      window.location.hash = `/editor/${res.project.id}`;
   233→   233→    },
   234→   234→    onError: (e: Error) =>
   235→   235→      toast({ title: 'خطا', description: e.message, variant: 'destructive' }),
   236→   236→  });
   237→   237→
   238→   238→  return (
   239→   239→    <Card className="w-full">
   240→   240→      <CardContent className="p-5 space-y-4">
   241→   241→        <div className="flex items-center gap-2">
   242→   242→          <Camera className="h-5 w-5 text-primary" />
   243→   243→          <h2 className="font-bold">بازسازی دقیق فضا از عکس یا ویدیو</h2>
   244→   244→        </div>
   245→   245→
   246→   246→        <input
   247→   247→          ref={imgRef}
   248→   248→          type="file"
   249→   249→          accept="image/*"
   250→   250→          multiple
   251→   251→          className="hidden"
   252→   252→          onChange={(e) => {
   253→   253→            addImages(e.target.files);
   254→   254→            e.target.value = '';
   255→   255→          }}
   256→   256→          aria-hidden
   257→   257→        />
   258→   258→        <input
   259→   259→          ref={vidRef}
   260→   260→          type="file"
   261→   261→          accept="video/*"

   265→   265→            e.target.value = '';
   266→   266→          }}
   267→   267→          aria-hidden
   268→   268→        />
   269→   269→
   270→   270→        <div
   271→   271→          role="button"
   272→   272→          tabIndex={0}
   273→   273→          onClick={() => imgRef.current?.click()}
   274→   274→          onKeyDown={(e) => {
   275→   275→            if (e.key === 'Enter' || e.key === ' ') {
   276→   276→              e.preventDefault();

   286→   286→          aria-label="انتخاب عکس‌های فضا"

   383→   383→function FolderDialog({

   387→   387→  onSave,
   388→   388→  saving,
   389→   389→}: {
   390→   390→  open: boolean;
   391→   391→  onOpenChange: (v: boolean) => void;
   392→   392→  initial?: { id?: string; name?: string };
   393→   393→  onSave: (name: string) => void;
   394→   394→  saving: boolean;
   395→   395→}) {
   396→   396→  const [name, setName] = useState(initial?.name ?? '');
   397→   397→
   398→   398→  return (
   399→   399→    <Dialog open={open} onOpenChange={onOpenChange}>
   400→   400→      <DialogContent dir="rtl" className="max-w-sm">
   401→   401→        <DialogHeader>
   402→   402→          <DialogTitle>{initial?.id ? 'تغییر نام پوشه' : 'پوشه جدید'}</DialogTitle>
   403→   403→        </DialogHeader>
   404→   404→        <Input
   405→   405→          value={name}
   406→   406→          onChange={(e) => setName(e.target.value)}
   407→   407→          placeholder="مثلاً: خانه آقای رضایی"
   408→   408→          maxLength={60}
   409→   409→          autoFocus
   410→   410→          onKeyDown={(e) => {
   411→   411→            if (e.key === 'Enter' && name.trim()) onSave(name.trim());
   412→   412→          }}
   413→   413→        />
   414→   414→        <DialogFooter>
   415→   415→          <Button variant="outline" onClick={() => onOpenChange(false)}>انصراف</Button>
   416→   416→          <Button onClick={() => name.trim() && onSave(name.trim())} disabled={!name.trim() || saving}>
   417→   417→            {saving && <Loader2 className="h-4 w-4 animate-spin" />} ذخیره
   418→   418→          </Button>
   419→   419→        </DialogFooter>
   420→   420→      </DialogContent>
   421→   421→    </Dialog>
   422→   422→  );
   423→   423→}
