    63→    63→    try {
    64→    64→      await api.login(pw);
    65→    65→      onDone();
    66→    66→    } catch (e) {

    68→    68→        title: 'ورود ناموفق',
    69→    69→        description: (e as Error).message || 'رمز مدیر اشتباه است',
    70→    70→        variant: 'destructive',
    71→    71→      });
    72→    72→    } finally {
    73→    73→      setBusy(false);
    74→    74→    }
    75→    75→  };

    83→    83→          </div>
    84→    84→          <div className="space-y-1.5">
