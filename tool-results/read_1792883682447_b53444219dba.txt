    90→          </div>
    91→          <div className="space-y-2.5">

   122→// ---------- کارت آپلود (چند عکس + ویدیو) ----------
   123→function UploadCard({ folderId }: { folderId: string | null }) {
   124→  const { toast } = useToast();
   125→  const qc = useQueryClient();
   126→  const imgRef = useRef<HTMLInputElement>(null);
   127→  const vidRef = useRef<HTMLInputElement>(null);
   128→  const [shots, setShots] = useState<Shot[]>([]);
   129→  const [videoKey, setVideoKey] = useState<string | null>(null);
   130→  const [name, setName] = useState('');
   131→  const [aiStep, setAiStep] = useState(-1);
   132→
   133→  useEffect(() => {
   134→    if (aiStep < 0) return;
   135→    const timer = setInterval(() => {
   136→      setAiStep((s) => Math.min(s + 1, AI_STEPS.length - 1));
   137→    }, 1800);
   138→    return () => clearInterval(timer);
   139→  }, [aiStep]);
   140→
   141→  const addImages = async (files: FileList | null) => {
   142→    if (!files?.length) return;
   143→    const imgs = Array.from(files).filter((f) => f.type.startsWith('image/'));
   144→    if (!imgs.length) {
   145→      toast({ title: 'فقط تصویر مجاز است', variant: 'destructive' });
   146→      return;
   147→    }
   148→    try {
   149→      const urls = await Promise.all(imgs.slice(0, 4).map((f) => fileToResizedDataUrl(f)));
   150→      setShots((prev) => {
   151→        const merged = [
   152→          ...prev.filter((s) => s.key !== videoKey || !videoKey),
   153→          ...urls.map((src, i) => ({ key: `img-${Date.now()}-${i}`, src })),
   154→        ];
   155→        return merged.slice(0, 5);
   156→      });
   157→    } catch {
   158→      toast({ title: 'خواندن تصویر ناموفق بود', variant: 'destructive' });
   159→    }
   160→  };
   161→
   162→  const addVideo = async (f: File | null | undefined) => {
   163→    if (!f) return;
   164→    if (!f.type.startsWith('video/')) {
   165→      toast({ title: 'فایل ویدیویی معتبر نیست', variant: 'destructive' });
   166→      return;
   167→    }
   168→    try {
   169→      toast({ title: 'در حال استخراج فریم‌های ویدیو...' });
   170→      const frames = await videoToFrames(f, 3);
   171→      const vk = `vid-${Date.now()}`;
   172→      setVideoKey(vk);
   173→      setShots((prev) =>
   174→        [...prev.filter((s) => !s.key.startsWith('vid-')), ...frames.map((src, i) => ({ key: `${vk}-${i}`, src }))].slice(0, 5)
   175→      );
   176→    } catch (e) {
   177→      toast({
   178→        title: 'استخراج فریم ناموفق بود',
   179→        description: e instanceof Error ? e.message : undefined,
   180→        variant: 'destructive',
   181→      });
   182→    }
   183→  };
   184→
   185→  const reconstruct = useMutation({
   186→    mutationFn: async () => {
   187→      if (!shots.length) throw new Error('حداقل یک عکس یا ویدیو انتخاب کنید');
   188→      setAiStep(0);
   189→      const res = await api.reconstruct(shots.map((s) => s.src));
   190→      await new Promise((r) => setTimeout(r, 800));
   191→      return res;
   192→    },
   193→    onSuccess: async (res) => {
   194→      setAiStep(AI_STEPS.length - 1);
   195→      const created = await api.createProject({
   196→        name: name.trim() || 'پروژه بازسازی‌شده',
   197→        roomType: res.roomType,
   198→        roomSpec: res.roomSpec,
   199→        sceneState: { objects: res.objects },
   200→        folderId,
   201→      });
   202→      qc.invalidateQueries({ queryKey: ['projects'] });
   203→      toast({
   204→        title: 'مدل سه‌بعدی ساخته شد ✓',
   205→        description: res.objects.length
   206→          ? `${res.objects.length} قطعه تجهیزات/کابینت موجود شناسایی شد`
   207→          : res.description || undefined,
   208→      });
   209→      window.location.hash = `/editor/${created.project.id}`;
   210→    },
   211→    onError: (e: Error) => {
   212→      setAiStep(-1);
   213→      toast({
   214→        title: 'بازسازی ناموفق بود',
   215→        description: `${e.message} — می‌توانید با فضای نمونه شروع کنید`,
   216→        variant: 'destructive',
   217→      });
   218→    },
   219→  });
   220→
   221→  const startSample = useMutation({
   222→    mutationFn: () =>
   223→      api.createProject({
   224→        name: name.trim() || 'آشپزخانه نمونه',
   225→        roomType: 'kitchen',
   226→        roomSpec: SAMPLE_ROOM_SPEC,
   227→        sceneState: SAMPLE_SCENE,
   228→        folderId,
   229→      }),
   230→    onSuccess: (res) => {
   231→      qc.invalidateQueries({ queryKey: ['projects'] });
   232→      window.location.hash = `/editor/${res.project.id}`;
   233→    },
   234→    onError: (e: Error) =>
   235→      toast({ title: 'خطا', description: e.message, variant: 'destructive' }),
   236→  });
   237→
   238→  return (
   239→    <Card className="w-full">
   240→      <CardContent className="p-5 space-y-4">
   241→        <div className="flex items-center gap-2">
   242→          <Camera className="h-5 w-5 text-primary" />
   243→          <h2 className="font-bold">بازسازی دقیق فضا از عکس یا ویدیو</h2>
   244→        </div>
   245→
   246→        <input
   247→          ref={imgRef}
   248→          type="file"
   249→          accept="image/*"
   250→          multiple
   251→          className="hidden"
   252→          onChange={(e) => {
   253→            addImages(e.target.files);
   254→            e.target.value = '';
   255→          }}
   256→          aria-hidden
   257→        />
   258→        <input
   259→          ref={vidRef}
   260→          type="file"
   261→          accept="video/*"
   262→          className="hidden"
   263→          onChange={(e) => {
   264→            addVideo(e.target.files?.[0]);
   265→            e.target.value = '';
   266→          }}
   267→          aria-hidden
   268→        />
   269→
   270→        <div
   271→          role="button"
   272→          tabIndex={0}
   273→          onClick={() => imgRef.current?.click()}
   274→          onKeyDown={(e) => {
   275→            if (e.key === 'Enter' || e.key === ' ') {
   276→              e.preventDefault();
   277→              imgRef.current?.click();
   278→            }
   279→          }}
   280→          onDragOver={(e) => e.preventDefault()}
   281→          onDrop={(e) => {
   282→            e.preventDefault();
   283→            addImages(e.dataTransfer.files);
   284→          }}
   285→          className="w-full rounded-xl border-2 border-dashed border-border hover:border-primary/60 bg-muted/30 transition-colors p-6 flex flex-col items-center gap-2 cursor-pointer group min-h-[120px] justify-center outline-none focus-visible:ring-2 focus-visible:ring-primary/50"
   286→          aria-label="انتخاب عکس‌های فضا"
   287→        >
   288→          {shots.length === 0 ? (
   289→            <>
   290→              <UploadCloud className="h-9 w-9 text-muted-foreground/50 group-hover:text-primary transition-colors" />
   291→              <p className="text-sm font-medium">چند عکس از زوایای مختلف بکشید یا انتخاب کنید</p>
   292→              <p className="text-xs text-muted-foreground">
   293→                روبه‌رو، چپ و راست — پوشش کامل ۳۶۰ درجه لازم نیست
   294→              </p>
   295→            </>
   296→          ) : (
   297→            <div className="grid grid-cols-3 gap-2 w-full">
   298→              {shots.map((s) => (
   299→                <div key={s.key} className="relative group/th rounded-lg overflow-hidden border">
   300→                  <img src={s.src} alt="نمای آپلودشده فضا" className="w-full h-16 object-cover" />
   301→                  <button
   302→                    type="button"
   303→                    onClick={(e) => {
   304→                      e.stopPropagation();
   305→                      setShots((prev) => prev.filter((x) => x.key !== s.key));
   306→                    }}
   307→                    className="absolute top-1 end-1 h-5 w-5 rounded-full bg-black/60 text-white flex items-center justify-center opacity-0 group-hover/th:opacity-100 transition-opacity"
   308→                    aria-label="حذف تصویر"
   309→                  >
   310→                    <X className="h-3 w-3" />
   311→                  </button>
   312→                </div>
   313→              ))}
   314→              <div className="h-16 rounded-lg border border-dashed flex flex-col items-center justify-center text-[10px] text-muted-foreground gap-0.5">
   315→                <ImageIcon className="h-4 w-4" />
   316→                افزودن (تا {5 - shots.length})
   317→              </div>
   318→            </div>
   319→          )}
   320→        </div>
   321→
   322→        <div className="grid grid-cols-2 gap-2">
   323→          <Button
   324→            type="button"
   325→            variant="outline"
   326→            size="sm"
   327→            className="h-9 gap-1.5 text-xs"
   328→            onClick={() => imgRef.current?.click()}
   329→            disabled={shots.length >= 5}
   330→          >
   331→            <ImageIcon className="h-3.5 w-3.5" /> عکس‌های بیشتر
   332→          </Button>
   333→          <Button
   334→            type="button"
   335→            variant="outline"
   336→            size="sm"
   337→            className="h-9 gap-1.5 text-xs"
   338→            onClick={() => vidRef.current?.click()}
   339→            disabled={shots.length >= 4}
   340→          >
   341→            <Video className="h-3.5 w-3.5" /> آپلود ویدیو
   342→          </Button>
   343→        </div>
   344→
   345→        <Input
   346→          value={name}
   347→          onChange={(e) => setName(e.target.value)}
   348→          placeholder="نام پروژه (اختیاری)"
   349→          maxLength={80}
   350→        />
   351→
   352→        <div className="flex flex-col sm:flex-row gap-2">
   353→          <Button
   354→            className="flex-1 gap-2 h-11"
   355→            onClick={() => reconstruct.mutate()}
   356→            disabled={!shots.length || reconstruct.isPending}
   357→          >
   358→            <Sparkles className="h-4 w-4" />
   359→            {reconstruct.isPending ? 'در حال بازسازی...' : 'بازسازی سه‌بعدی با AI'}
   360→          </Button>
   361→          <Button
   362→            variant="outline"
   363→            className="sm:w-auto gap-2 h-11"
   364→            onClick={() => startSample.mutate()}
   365→            disabled={startSample.isPending}
   366→          >
   367→            <Boxes className="h-4 w-4" />
   368→            فضای نمونه
   369→          </Button>
   370→        </div>
   371→        <p className="text-[11px] text-muted-foreground leading-relaxed">
   372→          AI دیوارها، بازشوها و تجهیزات موجود (کابینت، یخچال، اجاق، هود) را تشخیص می‌دهد و مدل
   373→          سه‌بعدی تقریبی اما قابل ویرایش می‌سازد.
   374→        </p>
   375→      </CardContent>
   376→
   377→      {aiStep >= 0 && <ReconstructOverlay step={aiStep} />}
   378→    </Card>
   379→  );
   380→}

   382→// ---------- دیالوگ پوشه ----------
   383→function FolderDialog({
   384→  open,
   385→  onOpenChange,
   386→  initial,
   387→  onSave,
   388→  saving,
   389→}: {
   390→  open: boolean;
   391→  onOpenChange: (v: boolean) => void;
   392→  initial?: { id?: string; name?: string };
   393→  onSave: (name: string) => void;
   394→  saving: boolean;
   395→}) {
   396→  const [name, setName] = useState(initial?.name ?? '');
   397→
   398→  return (
   399→    <Dialog open={open} onOpenChange={onOpenChange}>
   400→      <DialogContent dir="rtl" className="max-w-sm">
   401→        <DialogHeader>
   402→          <DialogTitle>{initial?.id ? 'تغییر نام پوشه' : 'پوشه جدید'}</DialogTitle>
   403→        </DialogHeader>
   404→        <Input
   405→          value={name}
   406→          onChange={(e) => setName(e.target.value)}
   407→          placeholder="مثلاً: خانه آقای رضایی"
   408→          maxLength={60}
   409→          autoFocus
   410→          onKeyDown={(e) => {
   411→            if (e.key === 'Enter' && name.trim()) onSave(name.trim());
   412→          }}
   413→        />
   414→        <DialogFooter>
   415→          <Button variant="outline" onClick={() => onOpenChange(false)}>انصراف</Button>
   416→          <Button onClick={() => name.trim() && onSave(name.trim())} disabled={!name.trim() || saving}>
   417→            {saving && <Loader2 className="h-4 w-4 animate-spin" />} ذخیره
   418→          </Button>
   419→        </DialogFooter>
   420→      </DialogContent>
   421→    </Dialog>
   422→  );
   423→}

   425→// ---------- نوار پوشه‌ها ----------
   426→function FolderBar({
   427→  folders,
   428→  selected,
   429→  onSelect,
   430→  currentFolderId,
   431→}: {
   432→  folders: FolderDTO[];
   433→  selected: FolderFilter;
   434→  onSelect: (f: FolderFilter) => void;
   435→  currentFolderId: string | null;
   436→}) {
   437→  const { toast } = useToast();
   438→  const qc = useQueryClient();
   439→  const [dlgOpen, setDlgOpen] = useState(false);
   440→  const [editing, setEditing] = useState<{ id?: string; name?: string } | undefined>();
   441→
   442→  const createFolder = useMutation({
   443→    mutationFn: (name: string) => api.createFolder(name),
   444→    onSuccess: (res) => {
   445→      qc.invalidateQueries({ queryKey: ['folders'] });
   446→      setDlgOpen(false);
   447→      onSelect({ kind: 'folder', id: res.folder.id });
   448→      toast({ title: `پوشه «${res.folder.name}» ساخته شد` });
   449→    },
   450→    onError: (e: Error) => toast({ title: 'خطا', description: e.message, variant: 'destructive' }),
   451→  });
   452→
   453→  const renameFolder = useMutation({
   454→    mutationFn: ({ id, name }: { id: string; name: string }) => api.renameFolder(id, name),
   455→    onSuccess: () => {
   456→      qc.invalidateQueries({ queryKey: ['folders'] });
   457→      qc.invalidateQueries({ queryKey: ['projects'] });
   458→      setDlgOpen(false);
   459→      toast({ title: 'نام پوشه تغییر کرد' });
   460→    },
   461→    onError: (e: Error) => toast({ title: 'خطا', description: e.message, variant: 'destructive' }),
   462→  });
   463→
   464→  const deleteFolder = useMutation({
   465→    mutationFn: (id: string) => api.deleteFolder(id),
   466→    onSuccess: () => {
   467→      qc.invalidateQueries({ queryKey: ['folders'] });
   468→      qc.invalidateQueries({ queryKey: ['projects'] });
   469→      onSelect({ kind: 'all' });
   470→      toast({ title: 'پوشه حذف شد', description: 'پروژه‌های آن بدون پوشه شدند' });
   471→    },
   472→    onError: (e: Error) => toast({ title: 'خطا', description: e.message, variant: 'destructive' }),
   473→  });
   474→
   475→  const chip = (active: boolean) =>
   476→    `h-9 px-3 rounded-full border text-xs font-medium flex items-center gap-1.5 transition-colors touch-manipulation ${
   477→      active
   478→        ? 'bg-primary text-primary-foreground border-primary'
   479→        : 'bg-card hover:bg-accent/60 border-border'
   480→    }`;
   481→
   482→  const newEmptyProject = useMutation({
   483→    mutationFn: () =>
   484→      api.createProject({
   485→        name: 'پروژه جدید',
   486→        roomSpec: {
   487→          widthM: 4, depthM: 3.4, heightM: 2.8,
   488→          wallColor: '#ece7dd', floorColor: '#a97c50', ceilingColor: '#fafafa',
   489→          windows: [], doorways: [],
   490→        },
   491→        folderId: currentFolderId,
   492→      }),
   493→    onSuccess: (res) => (window.location.hash = `/editor/${res.project.id}`),
   494→    onError: (e: Error) => toast({ title: 'خطا', description: e.message, variant: 'destructive' }),
   495→  });
   496→
   497→  return (
   498→    <div className="flex flex-wrap items-center gap-2">
   499→      <button type="button" className={chip(selected.kind === 'all')} onClick={() => onSelect({ kind: 'all' })}>
   500→        <Boxes className="h-3.5 w-3.5" /> همه
   501→      </button>
   502→      <button type="button" className={chip(selected.kind === 'none')} onClick={() => onSelect({ kind: 'none' })}>
   503→        <FolderIcon className="h-3.5 w-3.5 opacity-50" /> بدون پوشه
   504→      </button>
   505→      {folders.map((f) => (
   506→        <div key={f.id} className="flex items-center">
   507→          <button
   508→            type="button"
   509→            className={`${chip(selected.kind === 'folder' && selected.id === f.id)} rounded-s-full`}
   510→            onClick={() => onSelect({ kind: 'folder', id: f.id })}
   511→          >
   512→            <FolderIcon className="h-3.5 w-3.5" />
   513→            {f.name}
   514→            {typeof f.projectCount === 'number' && (
