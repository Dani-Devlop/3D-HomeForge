   236→/** عدد صحیح اختیاری در بازه — نامعتبر/غایب → undefined */
   237→function optInt(v: unknown, min: number, max: number): number | undefined {
   238→  const n = typeof v === 'number' ? v : typeof v === 'string' ? Number.parseFloat(v) : Number.NaN;
   239→  if (!Number.isFinite(n)) return undefined;
   240→  return Math.round(clamp(n, min, max));
   241→}

   245→/** خروجی مدل را به تحلیل امن تبدیل می‌کند؛ ورودی غیرآبجکت → null */
   246→function normalizeAnalysis(data: unknown): Analysis | null {
   247→  if (!isRecord(data)) return null;
   248→  const room = isRecord(data.room) ? data.room : {};
   249→
   250→  // پلان چندضلعی (دیوار کج) — نامعتبر → undefined (مستطیل)
   251→  const parsedOutline = normalizeOutline(room.outline);
   252→
   253→  // با outline معتبر، widthM/depthM باید دقیقاً bbox چندضلع باشند
   254→  let widthM = clamp(toFinite(room.widthM, 4), 2, 15);
   255→  let depthM = clamp(toFinite(room.depthM, 3.5), 2, 15);
   256→  if (parsedOutline) {
   257→    let minX = Infinity, maxX = -Infinity, minZ = Infinity, maxZ = -Infinity;
   258→    for (const [x, z] of parsedOutline) {
   259→      if (x < minX) minX = x;
   260→      if (x > maxX) maxX = x;
   261→      if (z < minZ) minZ = z;
   262→      if (z > maxZ) maxZ = z;
   263→    }
   264→    widthM = clamp(Math.round((maxX - minX) * 100) / 100, 2, 15);
   265→    depthM = clamp(Math.round((maxZ - minZ) * 100) / 100, 2, 15);
   266→  }
   267→
   268→  const windows = (Array.isArray(data.windows) ? data.windows : [])
   269→    .slice(0, 8)
   270→    .filter(isRecord)
   271→    .map((w) => ({
   272→      wall: toWall(w.wall),
   273→      ...(optInt(w.wallIndex, 0, 11) !== undefined ? { wallIndex: optInt(w.wallIndex, 0, 11) } : {}),
   274→      offsetX: clamp(toFinite(w.offsetX, 0), -7, 7),
   275→      sillY: clamp(toFinite(w.sillY, 1), 0.3, 1.8),
   276→      width: clamp(toFinite(w.width, 1.2), 0.6, 2.6),
   277→      height: clamp(toFinite(w.height, 1.2), 0.6, 2.2),
   278→    }));
   279→
   280→  const doorways = (Array.isArray(data.doorways) ? data.doorways : [])
   281→    .slice(0, 4)
   282→    .filter(isRecord)
   283→    .map((d) => ({
   284→      wall: toWall(d.wall),
   285→      ...(optInt(d.wallIndex, 0, 11) !== undefined ? { wallIndex: optInt(d.wallIndex, 0, 11) } : {}),
   286→      offsetX: clamp(toFinite(d.offsetX, 0), -7, 7),
   287→      width: clamp(toFinite(d.width, 0.95), 0.8, 1.4),
   288→      height: clamp(toFinite(d.height, 2.1), 1.9, 2.5),
   289→    }));
   290→
   291→  const items = (Array.isArray(data.items) ? data.items : [])
   292→    .slice(0, MAX_OBJECTS)
   293→    .filter(isRecord)
   294→    .map((it): ReconstructItem => {
   295→      const dRaw = toFinite(it.depthM, Number.NaN);
   296→      const wIdx = optInt(it.wallIndex, 0, 11);
   297→      return {
   298→        kind: toFurnitureKind(it.kind),
   299→        wall: toWall(it.wall),
   300→        ...(wIdx !== undefined ? { wallIndex: wIdx } : {}),
   301→        offsetX: clamp(toFinite(it.offsetX, 0), -7, 7),
   302→        widthM: clamp(toFinite(it.widthM, 0.8), 0.3, 3),
   303→        heightM: clamp(toFinite(it.heightM, 0.85), 0.3, 2.4),
   304→        depthM: Number.isFinite(dRaw) ? clamp(dRaw, 0.25, 1.2) : undefined,
   305→        glassDoors: it.glassDoors === true,
   306→        colorHex: toColor(it.colorHex, '#6b4f3a'),
   307→        description: typeof it.description === 'string' ? it.description.slice(0, 120) : '',
   308→      };
   309→    });
   310→
   311→  return {
   312→    roomType:
   313→      typeof data.roomType === 'string' && ROOM_TYPES.includes(data.roomType)
   314→        ? data.roomType
   315→        : 'other',
   316→    room: {
   317→      widthM,
   318→      depthM,
   319→      heightM: clamp(toFinite(room.heightM, 2.8), 2.2, 4.2),
   320→      ...(parsedOutline ? { outline: parsedOutline } : {}),
   321→      wallColor: toColor(room.wallColor, '#ece7dd'),
   322→      floorColor: toColor(room.floorColor, '#a97c50'),
   323→      ceilingColor: toColor(room.ceilingColor, '#fafafa'),
   324→    },
   325→    windows,
   326→    doorways,
   327→    items,
   328→    description: typeof data.description === 'string' ? data.description.slice(0, 600) : '',
   329→  };
   330→}

   332→/** نرمال‌سازی تدافعی خروجی نقشه — خطای نقشه هرگز کل درخواست را نمی‌شکند */
   333→function normalizeFloorPlan(data: unknown): FloorPlanSummary | null {
   334→  if (!isRecord(data)) return null;
   335→  return {
   336→    summary: typeof data.summary === 'string' ? data.summary.slice(0, 1200) : '',
   337→    widthM: optFinite(data.widthM, 2, 30),
   338→    depthM: optFinite(data.depthM, 2, 30),
   339→    ...(normalizeOutline(data.outline, { bound: 30 })
   340→      ? { outline: normalizeOutline(data.outline, { bound: 30 })! }
   341→      : {}),
   342→    rooms: (Array.isArray(data.rooms) ? data.rooms : [])
   343→      .slice(0, 20)
   344→      .filter(isRecord)
   345→      .map((r) => ({
   346→        name: typeof r.name === 'string' ? r.name.slice(0, 60) : '',
   347→        widthM: optFinite(r.widthM, 0.5, 30),
   348→        depthM: optFinite(r.depthM, 0.5, 30),
   349→      })),
   350→    windows: (Array.isArray(data.windows) ? data.windows : [])
   351→      .slice(0, 12)
   352→      .filter(isRecord),
   353→    notes: (Array.isArray(data.notes) ? data.notes : [])
   354→      .slice(0, 12)
   355→      .filter((n): n is string => typeof n === 'string')
   356→      .map((n) => n.slice(0, 200)),
   357→  };
   358→}

   379→/** تحلیل VLM یک تصویر با حداکثر ۲ تلاش */
   380→async function analyzeImage(zai: ZAI, image: string, index: number): Promise<Analysis | null> {
   381→  let lastError: unknown = null;
   382→  for (let attempt = 1; attempt <= 2; attempt++) {
   383→    try {
   384→      const visionResult = await zai.chat.completions.createVision(
   385→        visionBody(RECONSTRUCT_PROMPT, image),
   386→      );
   387→      const content = visionResult?.choices?.[0]?.message?.content;
   388→      if (typeof content !== 'string' || !content.trim()) {
   389→        throw new Error('پاسخ خالی از مدل');
   390→      }
   391→      const analysis = normalizeAnalysis(extractJsonObject(content));
   392→      if (!analysis) throw new Error('خروجی مدل آبجکت JSON معتبر نبود');
   393→      return analysis;
   394→    } catch (err) {
   395→      lastError = err;
   396→      console.error(`[reconstruct] vision attempt ${attempt} (image ${index}) failed:`, err);
   397→    }
   398→  }
   399→  console.error(`[reconstruct] all vision attempts failed for image ${index}:`, lastError);
   400→  return null;
   401→}

   409→    const content = visionResult?.choices?.[0]?.message?.content;
   410→    if (typeof content !== 'string' || !content.trim()) return null;
   411→    return normalizeFloorPlan(extractJsonObject(content));
   412→  } catch (err) {
   413→    console.error('[reconstruct] floor-plan analysis failed (ignored):', err);
   414→    return null;
   415→  }
   416→}
   417→
   418→/** ادغام تحلیل‌های جزئی (+ نقشه) با یک فراخوانی متنی؛ در صورت خطا اولین تحلیل برگردانده می‌شود */
   419→async function mergeAnalyses(
   420→  zai: ZAI,
   421→  partials: Analysis[],
   422→  floorPlan: FloorPlanSummary | null,
   423→): Promise<Analysis> {
   424→  if (partials.length === 1 && !floorPlan) return partials[0];
   425→  try {
   426→    // نقشه به‌عنوان ground truth همراه پارتیال‌ها داده می‌شود
   427→    const userContent = floorPlan
   428→      ? JSON.stringify({ floorPlan, parts: partials })
   429→      : JSON.stringify(partials);
   430→    const completion = await zai.chat.completions.create({
   431→      messages: [
   432→        { role: 'system', content: MERGE_PROMPT },
   433→        { role: 'user', content: userContent },
   434→      ] satisfies ChatMessage[],
   435→      thinking: { type: 'disabled' },
   436→    });
   437→    const content = completion?.choices?.[0]?.message?.content;
   438→    if (typeof content === 'string' && content.trim()) {
   439→      const merged = normalizeAnalysis(extractJsonObject(content));
   440→      if (merged) return merged;
   441→    }
   442→    throw new Error('خروجی ادغام نامعتبر بود');
   443→  } catch (err) {
   444→    console.error('[reconstruct] merge failed, using first partial:', err);
   445→    return partials[0];
   446→  }
   447→}

   475→/** روشنایی نسبی رنگ hex (۰ تا ۱) — برای جلوگیری از شیشه تیره */
   476→function luminance(hex: string): number {
   477→  const r = parseInt(hex.slice(1, 3), 16);
   478→  const g = parseInt(hex.slice(3, 5), 16);
   479→  const b = parseInt(hex.slice(5, 7), 16);
   480→  return (0.299 * r + 0.587 * g + 0.114 * b) / 255;
   481→}
   482→
   483→/** رنگ تیره را برای قاب شیشه به‌قدر کافی روشن می‌کند */
   484→const lightenIfDark = (hex: string): string =>
   485→  luminance(hex) < 0.35 ? shadeColor(hex, 0.65) : hex;
   486→
   487→/** آفست را طوری محدود می‌کند که آیتم داخل دیوار بماند */
   488→function clampOffsetOnWall(wall: WallName, offsetX: number, itemWidth: number, room: Analysis['room']) {
   489→  const wallLength = wall === 'back' || wall === 'front' ? room.widthM : room.depthM;
   490→  const maxAbs = wallLength / 2 - 0.05 - itemWidth / 2;
   491→  return maxAbs <= 0 ? 0 : clamp(offsetX, -maxAbs, maxAbs);
   492→}

   494→function buildObject(item: ReconstructItem, room: Analysis['room']): SceneObject {
   495→  const kind = item.kind;
   496→  const w = item.widthM;
   497→  const h = item.heightM;
   498→  const d = item.depthM ?? KIND_DEFAULT_D[kind];
   499→  const color = item.colorHex;
   500→
   501→  let rawParams: Record<string, unknown>;
   502→  if (kind === 'fridge') {
   503→    rawParams = {
   504→      w, h, d, color, bodyColor: color, doorColor: shadeColor(color, 0.2),
   505→      doors: 1, drawers: 0, handles: 'bar',
   506→    };
   507→  } else if (kind === 'stove') {
   508→    rawParams = {
   509→      w, h, d, color, bodyColor: color, doorColor: shadeColor(color, 0.15),
   510→      countertopColor: '#1e2024', doors: 1, drawers: 0, handles: 'bar',
   511→    };
   512→  } else if (kind === 'hood') {
   513→    rawParams = {
   514→      w, h, d, color, bodyColor: color,
   515→      doors: 0, drawers: 0, handles: 'none', mountY: 1.62,
   516→    };
   517→  } else if (kind === 'wall') {
   518→    rawParams = {
   519→      w, h, d, bodyColor: color, doorColor: shadeColor(color, 0.2),
   520→      countertopColor: '#d8d3c8', doors: w > 0.7 ? 2 : 1, drawers: 0,
   521→      handles: 'bar', mountY: 1.5,
   522→    };
   523→  } else if (kind === 'tv') {
   524→    // تلویزیون نازک — نصب روی دیوار با ارتفاع پیش‌فرض ۱.۴ متر
   525→    rawParams = { w, h, d: 0.06, color: '#101318', bodyColor: '#1b1f26', mountY: 1.4 };
   526→  } else if (kind === 'sofa') {
   527→    rawParams = { w, h: clamp(h, 0.7, 1.0), d: clamp(d, 0.8, 1.2), color, bodyColor: shadeColor(color, -0.15) };
   528→  } else if (kind === 'bed') {
   529→    rawParams = { w, h: 0.55, d: clamp(d, 1.8, 2.2), color, bodyColor: shadeColor(color, -0.2) };
   530→  } else if (kind === 'table') {
   531→    rawParams = { w, h: 0.75, d: clamp(d, 0.7, 1.2), color, bodyColor: shadeColor(color, -0.1) };
   532→  } else if (kind === 'chair') {
   533→    rawParams = { w: 0.45, h: 0.9, d: 0.5, color, bodyColor: color };
   534→  } else if (kind === 'washer') {
   535→    rawParams = { w: 0.6, h: 0.85, d: 0.6, color, bodyColor: color };
   536→  } else if (kind === 'dishwasher') {
   537→    rawParams = { w: 0.6, h: 0.85, d: 0.6, color, bodyColor: color, doors: 1, handles: 'bar' };
   538→  } else if (kind === 'microwave') {
   539→    // مایکروویو روی کانتر (y = 0.9)
   540→    rawParams = { w: clamp(w, 0.4, 0.7), h: 0.3, d: 0.38, color, bodyColor: color, mountY: 0.9 };
   541→  } else {
   542→    // base / tall / island / shelf
   543→    rawParams = {
   544→      w, h, d, bodyColor: color, doorColor: shadeColor(color, 0.2),
   545→      countertopColor: '#d8d3c8', doors: w > 0.7 ? 2 : 1, drawers: 0, handles: 'bar',
   546→    };
   547→  }
   548→
   549→  // درب شیشه‌ای — فقط انواع کابینتی؛ قاب شیشه هرگز تیره نمی‌ماند
   550→  if (GLASS_KINDS.has(kind)) {
   551→    rawParams.glass = item.glassDoors === true;
   552→    if (item.glassDoors) {
   553→      if (typeof rawParams.bodyColor === 'string') rawParams.bodyColor = lightenIfDark(rawParams.bodyColor);
   554→      if (typeof rawParams.doorColor === 'string') rawParams.doorColor = lightenIfDark(rawParams.doorColor);
   555→    }
   556→  }
   557→
   558→  const params = normalizePartialParams(rawParams, kind);
   559→  const y = kind === 'wall' || kind === 'hood' || kind === 'tv' || kind === 'microwave'
   560→    ? params.mountY ?? 0
   561→    : 0;
   562→
   563→  // جانمایی: لبه کج outline (wallIndex) در اولویت، وگرنه دیوار نامی مستطیل
   564→  let position: [number, number, number];
   565→  let rotationY: number;
   566→  if (room.outline && room.outline.length >= 3 && item.wallIndex !== undefined) {
   567→    const edge = edgeFrame(room.outline, item.wallIndex);
   568→    const maxAbs = Math.max(0, edge.len / 2 - 0.05 - w / 2);
   569→    const off = maxAbs <= 0 ? 0 : clamp(item.offsetX, -maxAbs, maxAbs);
   570→    ({ position, rotationY } = placeAtEdge(room.outline, item.wallIndex, off, d, y));
   571→  } else {
   572→    const offsetX = clampOffsetOnWall(item.wall, item.offsetX, w, room);
   573→    ({ position, rotationY } = placeAtWall(
   574→      item.wall,
   575→      offsetX,
   576→      d,
   577→      { widthM: room.widthM, depthM: room.depthM },
   578→      y,
   579→    ));
   580→  }
   581→
   582→  return {
   583→    id: crypto.randomUUID().slice(0, 8),
   584→    type: kind as ObjectType,
   585→    label: RECONSTRUCT_LABELS[kind],
   586→    params,
   587→    position,
   588→    rotationY,
   589→    ...(kind === 'tv' ? { on: false } : {}),
   590→  };
   591→}

   595→interface ReconstructResult {
   596→  roomType: string;
   597→  roomSpec: RoomSpec;
   598→  objects: SceneObject[];
   599→  description: string;

   602→/**
   603→ * خط لوله کامل: تحلیل موازی همه عکس‌ها + نقشه → ادغام → ساخت آبجکت‌ها.
   604→ * send در حالت استریم رویدادهای progress را می‌نویسد؛ در حالت JSON undefined است.
   605→ */
   606→async function runPipeline(
   607→  zai: ZAI,
   608→  images: string[],
   609→  floorPlanImage: string | undefined,
   610→  send?: SendFn,
   611→): Promise<ReconstructResult> {
   612→  // نقشه در همان دسته موازی اجرا می‌شود؛ خطایش نادیده گرفته می‌شود
   613→  const floorPlanPromise: Promise<FloorPlanSummary | null> | null = floorPlanImage
   614→    ? (async () => {
   615→        send?.({ type: 'progress', stage: 'floorplan', message: 'در حال بررسی نقشه ساختمان...' });
   616→        const fp = await analyzeFloorPlan(zai, floorPlanImage);
   617→        if (fp) {
   618→          send?.({ type: 'progress', stage: 'floorplan', message: 'نقشه ساختمان بررسی شد' });
   619→        }
   620→        return fp;
   621→      })()
   622→    : null;
   623→
   624→  // تحلیل موازی تصاویر — بعد از اتمام هر عکس یک رویداد progress می‌آید
   625→  const total = images.length;
   626→  const imagePromises = images.map((img, i) =>
   627→    analyzeImage(zai, img, i).then((analysis) => {
   628→      send?.({
   629→        type: 'progress',
   630→        stage: 'analyze',
   631→        index: i + 1,
   632→        total,
   633→        message: `تحلیل عکس ${faDigits(i + 1)} از ${faDigits(total)} انجام شد`,
   634→      });
   635→      return analysis;
   636→    }),
   637→  );
   638→
   639→  // همه در یک Promise.allSettled — نقشه اول صف است (اگر بوده باشد)
   640→  const tasks: Promise<unknown>[] = [
   641→    ...(floorPlanPromise ? [floorPlanPromise] : []),
   642→    ...imagePromises,
   643→  ];
   644→  const settled = await Promise.allSettled(tasks);
   645→  const offset = floorPlanPromise ? 1 : 0;
   646→
   647→  const floorPlanSummary = floorPlanPromise
   648→    ? settled[0].status === 'fulfilled'
   649→      ? (settled[0].value as FloorPlanSummary | null)
   650→      : null
   651→    : null;
   652→
   653→  const partials = settled
   654→    .slice(offset)
   655→    .map((r) => (r.status === 'fulfilled' ? (r.value as Analysis | null) : null))
   656→    .filter((a): a is Analysis => !!a);
   657→
   658→  if (partials.length === 0) {
   659→    throw new ReconstructFailure('تحلیل تصاویر با هوش مصنوعی ناموفق بود');
   660→  }
   661→
   662→  // مرحله ادغام (با نقشه به‌عنوان ground truth)
   663→  if (partials.length > 1) {
   664→    send?.({ type: 'progress', stage: 'merge', message: 'ادغام نتایج تحلیل‌ها...' });
   665→  }
   666→  const merged = await mergeAnalyses(zai, partials, floorPlanSummary);
   667→
   668→  const roomSpec: RoomSpec = {
   669→    ...merged.room,
   670→    windows: merged.windows,
   671→    doorways: merged.doorways,
   672→  };
   673→  const objects = merged.items.map((item) => buildObject(item, merged.room));
   674→
   675→  return {
   676→    roomType: merged.roomType,
   677→    roomSpec,
   678→    objects,
   679→    description: merged.description,
   680→  };
   681→}

   685→function validateImage(image: string): NextResponse | null {
   686→  if (image.length > MAX_IMAGE_LENGTH) {
   687→    return NextResponse.json(
   688→      { error: 'حجم یکی از تصاویر بیش از حد مجاز است (حداکثر ۷ مگابایت)' },
   689→      { status: 400 },
   690→    );
   691→  }
   692→  if (!DATA_URL_RE.test(image) && !RAW_BASE64_RE.test(image)) {
   693→    return NextResponse.json(
   694→      { error: 'فرمت یکی از ارسالی‌ها تصویر نیست' },
   695→      { status: 400 },
   696→    );
   697→  }
   698→  return null;
   699→}

   701→// ---------------- استریم NDJSON ----------------
   702→
   703→function ndjsonResponse(run: (send: SendFn) => Promise<void>): NextResponse {
   704→  const encoder = new TextEncoder();
   705→  const stream = new ReadableStream<Uint8Array>({
   706→    async start(controller) {
   707→      let closed = false;
   708→      const send: SendFn = (obj) => {
   709→        if (closed) return;
   710→        try {
   711→          controller.enqueue(encoder.encode(JSON.stringify(obj) + '\n'));
   712→        } catch {
   713→          // کلاینت قطع شده — بی‌خطر
   714→        }
   715→      };
   716→      try {
   717→        await run(send);
   718→      } catch (err) {
   719→        const message =
   720→          err instanceof ReconstructFailure
   721→            ? err.message
   722→            : 'بازسازی با هوش مصنوعی ناموفق بود';
   723→        if (!(err instanceof ReconstructFailure)) {
   724→          console.error('[reconstruct] stream failed:', err);
   725→        }
   726→        send({ type: 'error', error: message });
   727→      } finally {
   728→        closed = true;
   729→        try {
   730→          controller.close();
   731→        } catch {
   732→          // قبلاً بسته شده
   733→        }
   734→      }
   735→    },
   736→  });
   737→  return new NextResponse(stream, {
   738→    headers: {
   739→      'content-type': 'application/x-ndjson',
   740→      'cache-control': 'no-cache, no-transform',
   741→    },
   742→  });
   743→}

   747→export async function POST(req: NextRequest) {
   748→  // گارد مالک — قبل از هر چیز (AI گران است)
   749→  const denied = await requireOwner(req);
   750→  if (denied) return denied;
   751→
   752→  // حالت JSON (سازگار با نسخه ۲) در صورت Accept: application/json؛ پیش‌فرض استریم NDJSON
   753→  const wantsJson = (req.headers.get('accept') ?? '').includes('application/json');
   754→
   755→  const body = await readJsonBody(req);
   756→  if (!body.ok) {
   757→    return NextResponse.json({ error: 'داده ارسالی نامعتبر است' }, { status: 400 });
   758→  }
   759→
   760→  try {
   761→    const parsed = bodySchema.safeParse(body.data);
   762→    if (!parsed.success) {
   763→      return NextResponse.json(
   764→        { error: `حداکثر ${MAX_IMAGES} تصویر و حداقل یک تصویر لازم است` },
   765→        { status: 400 },
   766→      );
   767→    }
   768→
   769→    // سازگاری با نسخه ۱: { image } → { images: [image] }
   770→    const images = (parsed.data.images ?? (parsed.data.image ? [parsed.data.image] : []))
   771→      .map((s) => s.trim())
   772→      .filter(Boolean);
   773→    if (images.length === 0) {
   774→      return NextResponse.json({ error: 'تصویری ارسال نشده است' }, { status: 400 });
   775→    }
   776→
   777→    // اعتبارسنجی همه تصاویر — هر تصویر نامعتبر کل درخواست را رد می‌کند
   778→    for (const image of images) {
   779→      const invalid = validateImage(image);
   780→      if (invalid) return invalid;
   781→    }
   782→    const floorPlan = parsed.data.floorPlan?.trim() || undefined;
   783→    if (floorPlan) {
   784→      const invalid = validateImage(floorPlan);
   785→      if (invalid) return invalid;
   786→    }
   787→
   788→    const zai = await ZAI.create();
   789→
   790→    if (wantsJson) {
   791→      try {
   792→        const result = await runPipeline(zai, images, floorPlan);
   793→        return NextResponse.json(result);
   794→      } catch (err) {
   795→        if (err instanceof ReconstructFailure) {
   796→          return NextResponse.json({ error: err.message }, { status: 502 });
   797→        }
   798→        throw err;
   799→      }
   800→    }
   801→
   802→    // حالت استریم — رویدادهای progress و در پایان result یا error
   803→    return ndjsonResponse(async (send) => {
   804→      const result = await runPipeline(zai, images, floorPlan, send);
   805→      send({ type: 'result', ...result });
   806→    });
   807→  } catch (err) {
   808→    console.error('POST /api/reconstruct failed:', err);
   809→    return NextResponse.json(
   810→      { error: 'بازسازی با هوش مصنوعی ناموفق بود' },
   811→      { status: 502 },
   812→    );
   813→  }
   814→}
